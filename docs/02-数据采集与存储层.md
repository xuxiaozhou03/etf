# 02-数据采集与存储层

## 📖 文档概述

本文档详细描述量化回测系统的**数据采集与存储层**，涵盖从前复权API获取数据、ETF列表获取与过滤、到SQLite数据库持久化的完整链路。

**目标读者：** 数据开发工程师

## 1. 概述

### 1.1 数据层职责

| 职责            | 说明                                                                       |
| :-------------- | :------------------------------------------------------------------------- |
| **K线数据采集** | 调用前复权API获取日K线数据，带重试和限流处理                               |
| **ETF列表采集** | 调用ETF列表API，按规则过滤（trackingIndex≠null + 规模>3亿 + 同指数取最大） |
| **数据存储**    | 使用SQLite + Prisma持久化存储日K线和阶段收益                               |
| **数据同步**    | 支持增量/全量同步，避免重复拉取                                            |

### 1.2 数据流

```
┌─────────────────────────────────────────────────────────────────┐
│                  1. 获取ETF列表                                 │
│  ETF API → 所有ETF → 过滤规则                                   │
│  (trackingIndex≠null + 规模>3亿 + 同指数取最大)                 │
└────────────────────────────┬────────────────────────────────────┘
                             │
                             ▼
┌─────────────────────────────────────────────────────────────────┐
│                  2. 存储到数据库                                │
│  Security表 → 保存过滤后的ETF信息                               │
└────────────────────────────┬────────────────────────────────────┘
                             │
                             ▼
┌─────────────────────────────────────────────────────────────────┐
│                  3. 同步K线数据                                 │
│  对每个ETF → K线API (adjust=1) → 存储到Kline表                 │
└────────────────────────────┬────────────────────────────────────┘
                             │
                             ▼
┌─────────────────────────────────────────────────────────────────┐
│                  4. 计算阶段收益                                │
│  对每个ETF → 周/月/季/年/YTD收益 → 存储到PeriodReturn表         │
└─────────────────────────────────────────────────────────────────┘
```

### 3.4 采集器实现（带重试和限流）

```typescript
// packages/data/src/data-fetcher.ts

import pRetry from "p-retry";
import { Result, ok, err } from "@quant-backtest/core";

export interface RawKlineItem {
  week: string;
  tradeDate: string;
  open: number;
  high: number;
  low: number;
  close: number;
  volume: number;
  amount: number;
  change: number;
  changePercent: number;
}

export interface DataFetchError {
  type: "API_ERROR" | "NETWORK_ERROR" | "PARSE_ERROR";
  code?: string;
  message: string;
}

/**
 * 从API获取K线数据（带自动重试）
 */
export async function fetchKlineData(
  code: string,
  beginDate?: string,
  count: number = -1000,
): Promise<Result<RawKlineItem[], DataFetchError>> {
  const url = new URL("https://hongsehuojian.com/fundex-quote/line/kline");
  url.searchParams.set("securityCode", code);
  url.searchParams.set("count", String(count));
  url.searchParams.set("period", "day");
  url.searchParams.set("adjust", "1");
  if (beginDate) {
    url.searchParams.set("begin", beginDate);
  }

  try {
    const response = await pRetry(
      async () => {
        const res = await fetch(url.toString(), {
          headers: {
            accept: "application/json, text/plain, */*",
            "accept-language": "zh-CN,zh;q=0.9,en;q=0.8",
            "cache-control": "no-cache",
            pro: "RedRocket-PC",
          },
          method: "GET",
        });

        // 处理限流（429）
        if (res.status === 429) {
          const waitTime = parseInt(res.headers.get("retry-after") || "5000");
          console.warn(`[RateLimit] ${code} 触发限流，等待 ${waitTime}ms`);
          await new Promise((resolve) => setTimeout(resolve, waitTime));
          throw new Error("Rate limited, retrying...");
        }

        if (!res.ok) {
          throw new Error(`HTTP ${res.status}: ${res.statusText}`);
        }

        return res;
      },
      {
        retries: 3,
        minTimeout: 1000,
        factor: 2,
        onFailedAttempt: (error) => {
          console.warn(
            `[Fetch] ${code} 请求失败 (尝试 ${error.attemptNumber}/${error.retriesLeft + error.attemptNumber}): ${error.message}`,
          );
        },
      },
    );

    const json = await response.json();

    if (json.code !== "200") {
      return err({
        type: "API_ERROR",
        code: json.code,
        message: json.msg || "Unknown API error",
      });
    }

    const items = parseKlineResponse(json.data);
    return ok(items);
  } catch (error) {
    return err({
      type: "NETWORK_ERROR",
      message: error instanceof Error ? error.message : String(error),
      code: "FETCH_FAILED",
    });
  }
}

/**
 * 解析API返回数据
 */
export function parseKlineResponse(data: {
  securityCode: string;
  columns: string;
  items: string;
}): RawKlineItem[] {
  if (data.items === "") return [];
  const columnList = data.columns.split(",");

  return data.items.split(";").map((item) => {
    const values = item.split(",");
    return {
      week: values[0] || "",
      tradeDate: values[1] || "",
      open: parseFloat(values[2]) || 0,
      high: parseFloat(values[3]) || 0,
      low: parseFloat(values[4]) || 0,
      close: parseFloat(values[5]) || 0,
      volume: parseFloat(values[6]) || 0,
      amount: parseFloat(values[7]) || 0,
      change: parseFloat(values[8]) || 0,
      changePercent: parseFloat(values[9]) || 0,
    };
  });
}

/**
 * 日期转换：YYYY-MM-DD → YYYYMMDD
 */
export function parseDateToInt(dateStr: string): number {
  return parseInt(dateStr.replace(/-/g, ""), 10);
}

/**
 * 日期递增：YYYYMMDD → YYYYMMDD + 1
 */
export function incrementDate(date: number): string {
  const d = new Date(
    Math.floor(date / 10000),
    Math.floor((date % 10000) / 100) - 1,
    date % 100,
  );
  d.setDate(d.getDate() + 1);
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${year}${month}${day}`;
}
```

## 4. ETF列表采集

### 4.1 API接口

| 参数     | 值                                                             | 说明        |
| :------- | :------------------------------------------------------------- | :---------- |
| URL      | `https://hongsehuojian.com/fundex-quote/allPage/findListByEtf` | ETF列表接口 |
| orderBy  | `l.scale`                                                      | 按规模排序  |
| order    | `desc`                                                         | 降序        |
| pageNo   | `1`                                                            | 页码        |
| pageSize | `100`                                                          | 每页数量    |

### 4.2 过滤规则

| 步骤 | 过滤条件                                     | 说明                            |
| :--- | :------------------------------------------- | :------------------------------ |
| 1    | `trackingIndex !== null`                     | 过滤掉货币ETF等无跟踪指数的产品 |
| 2    | `scale >= 3亿`                               | 只保留规模大于3亿的ETF          |
| 3    | 按 `trackingIndex` 分组，保留 `scale` 最大的 | 同一指数只留规模最大的ETF       |

### 4.3 类别推断规则

| 跟踪指数关键词                                 | 类别 |
| :--------------------------------------------- | :--- |
| 沪深300、中证500、科创50、创业板、上证50       | 宽基 |
| 黄金                                           | 商品 |
| 转债、可交换、政金债、城投债                   | 债券 |
| 证券公司、证券、半导体、芯片、通信、医疗、医药 | 行业 |
| 其他                                           | 其他 |

### 4.4 ETF采集器实现

```typescript
// packages/data/src/etf-fetcher.ts

import pRetry from "p-retry";
import { Result, ok, err } from "@quant-backtest/core";

export interface RawETFItem {
  securityName: string;
  securityCode: string;
  scale: number;
  trackingIndex: string | null;
  managementFee: string;
  price: string;
  changePercent: string;
  managementcomp: string;
  fundManager: string;
}

export interface FilteredETF {
  code: string;
  name: string;
  market: "SH" | "SZ";
  category: string;
  trackingIndex: string;
  scale: number;
}

export interface ETFFilterOptions {
  minScale?: number; // 默认3亿
  filterNullTracking?: boolean; // 默认true
  keepLargestByIndex?: boolean; // 默认true
  pageSize?: number; // 默认100
}

export class ETFFetcher {
  private baseUrl =
    "https://hongsehuojian.com/fundex-quote/allPage/findListByEtf";

  /**
   * 获取所有ETF（分页遍历）
   */
  async fetchAllETFs(options?: {
    pageSize?: number;
  }): Promise<Result<RawETFItem[], string>> {
    const pageSize = options?.pageSize || 100;
    let allData: RawETFItem[] = [];
    let pageNo = 1;
    let total = 0;

    try {
      do {
        const result = await this.fetchPage(pageNo, pageSize);
        if (!result.ok) {
          return err(result.error);
        }

        const { data, total: totalCount } = result.value;
        allData = allData.concat(data);
        total = totalCount;
        pageNo++;

        console.log(`📊 已获取 ${allData.length}/${total} 条ETF数据`);
      } while (allData.length < total);

      return ok(allData);
    } catch (error) {
      return err(error instanceof Error ? error.message : String(error));
    }
  }

  /**
   * 获取单页ETF数据（带重试）
   */
  private async fetchPage(
    pageNo: number,
    pageSize: number,
  ): Promise<Result<{ data: RawETFItem[]; total: number }, string>> {
    const url = new URL(this.baseUrl);
    url.searchParams.set("orderBy", "l.scale");
    url.searchParams.set("order", "desc");
    url.searchParams.set("pageNo", String(pageNo));
    url.searchParams.set("pageSize", String(pageSize));

    try {
      const response = await pRetry(
        async () => {
          const res = await fetch(url.toString(), {
            headers: {
              accept: "application/json, text/plain, */*",
              "accept-language": "zh-CN,zh;q=0.9,en;q=0.8",
              "cache-control": "no-cache",
              pro: "RedRocket-PC",
            },
            method: "GET",
          });

          if (res.status === 429) {
            const waitTime = parseInt(res.headers.get("retry-after") || "5000");
            await new Promise((resolve) => setTimeout(resolve, waitTime));
            throw new Error("Rate limited");
          }

          if (!res.ok) {
            throw new Error(`HTTP ${res.status}`);
          }

          return res;
        },
        { retries: 3, minTimeout: 1000 },
      );

      const json = await response.json();

      if (json.code !== "200") {
        return err(`API错误: ${json.msg}`);
      }

      return ok({
        data: json.data.data,
        total: json.data.total,
      });
    } catch (error) {
      return err(error instanceof Error ? error.message : String(error));
    }
  }

  /**
   * 获取并过滤ETF
   */
  async fetchAndFilter(
    options: ETFFilterOptions = {},
  ): Promise<Result<FilteredETF[], string>> {
    const {
      minScale = 3,
      filterNullTracking = true,
      keepLargestByIndex = true,
      pageSize = 100,
    } = options;

    // 1. 获取所有ETF
    const fetchResult = await this.fetchAllETFs({ pageSize });
    if (!fetchResult.ok) {
      return err(fetchResult.error);
    }

    const allETFs = fetchResult.value;
    console.log(`📊 总ETF数量: ${allETFs.length}`);

    // 2. 应用过滤条件
    let filtered = allETFs;

    if (filterNullTracking) {
      filtered = filtered.filter((etf) => etf.trackingIndex !== null);
      console.log(`✅ 过滤trackingIndex为null后: ${filtered.length}`);
    }

    const minScaleValue = minScale * 1e8;
    filtered = filtered.filter((etf) => etf.scale >= minScaleValue);
    console.log(`✅ 规模 > ${minScale}亿: ${filtered.length}`);

    if (keepLargestByIndex) {
      const indexMap = new Map<string, RawETFItem>();
      for (const etf of filtered) {
        const key = etf.trackingIndex!;
        if (!indexMap.has(key) || etf.scale > indexMap.get(key)!.scale) {
          indexMap.set(key, etf);
        }
      }
      filtered = Array.from(indexMap.values());
      console.log(`✅ 每个跟踪指数只保留规模最大: ${filtered.length}`);
    }

    // 3. 转换为FilteredETF格式
    const result = filtered.map((etf) => ({
      code: etf.securityCode,
      name: etf.securityName,
      market: this.parseMarket(etf.securityCode),
      category: this.inferCategory(etf.trackingIndex || ""),
      trackingIndex: etf.trackingIndex || "",
      scale: etf.scale,
    }));

    return ok(result);
  }

  private parseMarket(code: string): "SH" | "SZ" {
    if (code.endsWith(".SH")) return "SH";
    if (code.endsWith(".SZ")) return "SZ";
    if (code.startsWith("51")) return "SH";
    if (code.startsWith("15") || code.startsWith("16")) return "SZ";
    return "SH";
  }

  private inferCategory(trackingIndex: string): string {
    if (!trackingIndex) return "其他";
    const index = trackingIndex.toLowerCase();

    if (index.includes("沪深300") || index.includes("000300")) return "宽基";
    if (index.includes("中证500") || index.includes("000905")) return "宽基";
    if (index.includes("科创50") || index.includes("000688")) return "宽基";
    if (index.includes("创业板") || index.includes("399006")) return "宽基";
    if (index.includes("上证50") || index.includes("000016")) return "宽基";
    if (index.includes("黄金")) return "商品";
    if (index.includes("转债") || index.includes("可交换")) return "债券";
    if (index.includes("政金债") || index.includes("城投债")) return "债券";
    if (index.includes("证券公司") || index.includes("证券")) return "行业";
    if (index.includes("半导体") || index.includes("芯片")) return "行业";
    if (index.includes("通信")) return "行业";
    if (index.includes("医疗") || index.includes("医药")) return "行业";
    return "其他";
  }
}
```

## 5. 数据写入服务

```typescript
// packages/data/src/data-writer.ts

import { PrismaClient } from "@prisma/client";
import { Result, ok, err } from "@quant-backtest/core";
import {
  fetchKlineData,
  parseDateToInt,
  incrementDate,
  DataFetchError,
} from "./data-fetcher";
import { ETFFetcher, FilteredETF, ETFFilterOptions } from "./etf-fetcher";
import { PeriodReturnCalculator } from "./period-return-calculator";

export interface SyncResult {
  code: string;
  name: string;
  klineInserted: number;
  klineUpdated: number;
  periodCount: number;
  isIncremental: boolean;
}

export class DataWriter {
  private prisma: PrismaClient;
  private periodCalculator: PeriodReturnCalculator;

  constructor(prisma?: PrismaClient) {
    this.prisma = prisma || new PrismaClient();
    this.periodCalculator = new PeriodReturnCalculator(this.prisma);
  }

  // ============================================================
  // K线同步
  // ============================================================

  /**
   * 同步单只ETF（支持全量/增量）
   */
  async syncETF(
    code: string,
    name: string,
    options: {
      market?: string;
      category?: string;
      incremental?: boolean;
      beginDate?: string;
    } = {},
  ): Promise<Result<SyncResult, string>> {
    const {
      market = "SH",
      category = "宽基",
      incremental = true,
      beginDate,
    } = options;

    console.log(
      `📊 同步 ${code} ${name} (${incremental ? "增量" : "全量"})...`,
    );

    try {
      // 1. 确保Security记录存在
      await this.prisma.security.upsert({
        where: { code },
        update: { name, market, category },
        create: { code, name, market, category, type: "etf" },
      });

      // 2. 确定开始日期
      let startDate: string | undefined = beginDate;
      if (incremental && !startDate) {
        const latest = await this.prisma.kline.findFirst({
          where: { code },
          orderBy: { date: "desc" },
          select: { date: true },
        });
        if (latest) {
          startDate = incrementDate(latest.date);
        }
      }

      // 3. 获取K线数据
      const fetchResult = await fetchKlineData(code, startDate);
      if (!fetchResult.ok) {
        return err(`获取数据失败: ${fetchResult.error.message}`);
      }

      const items = fetchResult.value;
      if (items.length === 0) {
        console.log(`⚠️ ${code} 无新增数据`);
        return ok({
          code,
          name,
          klineInserted: 0,
          klineUpdated: 0,
          periodCount: 0,
          isIncremental: incremental,
        });
      }

      // 4. 存储K线
      let inserted = 0;
      let updated = 0;

      for (const item of items) {
        const date = parseDateToInt(item.tradeDate);
        try {
          const result = await this.prisma.kline.upsert({
            where: { code_date: { code, date } },
            update: {
              week: item.week,
              open: item.open,
              high: item.high,
              low: item.low,
              close: item.close,
              volume: item.volume,
              amount: item.amount,
              change: item.change,
              changePercent: item.changePercent,
            },
            create: {
              code,
              date,
              week: item.week,
              open: item.open,
              high: item.high,
              low: item.low,
              close: item.close,
              volume: item.volume,
              amount: item.amount,
              change: item.change,
              changePercent: item.changePercent,
            },
          });
          if (result.createdAt.getTime() === result.updatedAt.getTime()) {
            inserted++;
          } else {
            updated++;
          }
        } catch (error) {
          console.error(`存储失败: ${code} ${date}`, error);
        }
      }

      console.log(`✅ K线: 新增 ${inserted} 条, 更新 ${updated} 条`);

      // 5. 全量重算阶段收益
      console.log(`📈 重算阶段收益...`);
      const periodResult =
        await this.periodCalculator.calculateAndSaveAll(code);
      const totalPeriods = Object.values(periodResult).reduce(
        (a, b) => a + b,
        0,
      );
      console.log(`✅ 阶段收益: ${totalPeriods} 条记录`);

      return ok({
        code,
        name,
        klineInserted: inserted,
        klineUpdated: updated,
        periodCount: totalPeriods,
        isIncremental: incremental,
      });
    } catch (error) {
      return err(error instanceof Error ? error.message : String(error));
    }
  }

  /**
   * 批量同步ETF
   */
  async syncETFs(
    etfs: Array<{
      code: string;
      name: string;
      market?: string;
      category?: string;
    }>,
    options?: { incremental?: boolean; beginDate?: string },
  ): Promise<{
    success: Array<{ code: string; name: string }>;
    failed: Array<{ code: string; error: string }>;
    summary: {
      totalKlineInserted: number;
      totalKlineUpdated: number;
      totalPeriodCount: number;
    };
  }> {
    const success: Array<{ code: string; name: string }> = [];
    const failed: Array<{ code: string; error: string }> = [];
    let totalKlineInserted = 0;
    let totalKlineUpdated = 0;
    let totalPeriodCount = 0;

    for (const etf of etfs) {
      const result = await this.syncETF(etf.code, etf.name, {
        market: etf.market,
        category: etf.category,
        ...options,
      });

      if (result.ok) {
        success.push({ code: etf.code, name: etf.name });
        totalKlineInserted += result.value.klineInserted;
        totalKlineUpdated += result.value.klineUpdated;
        totalPeriodCount += result.value.periodCount;
      } else {
        failed.push({ code: etf.code, error: result.error });
        console.error(`❌ ${etf.code} 同步失败:`, result.error);
      }
    }

    console.log(`
📊 批量同步完成:
  成功: ${success.length} 个
  失败: ${failed.length} 个
  K线新增: ${totalKlineInserted} 条
  K线更新: ${totalKlineUpdated} 条
  阶段收益: ${totalPeriodCount} 条
    `);

    return {
      success,
      failed,
      summary: {
        totalKlineInserted,
        totalKlineUpdated,
        totalPeriodCount,
      },
    };
  }

  // ============================================================
  // ETF列表同步
  // ============================================================

  /**
   * 同步ETF列表并存入数据库
   */
  async syncETFList(options: ETFFilterOptions = {}): Promise<
    Result<
      {
        total: number;
        inserted: number;
        updated: number;
        etfs: FilteredETF[];
      },
      string
    >
  > {
    const fetcher = new ETFFetcher();

    const fetchResult = await fetcher.fetchAndFilter(options);
    if (!fetchResult.ok) {
      return err(fetchResult.error);
    }

    const etfs = fetchResult.value;
    console.log(`📊 最终选中 ${etfs.length} 只ETF`);

    let inserted = 0;
    let updated = 0;

    for (const etf of etfs) {
      try {
        const result = await this.prisma.security.upsert({
          where: { code: etf.code },
          update: {
            name: etf.name,
            market: etf.market,
            category: etf.category,
            trackingIndex: etf.trackingIndex,
            scale: etf.scale,
          },
          create: {
            code: etf.code,
            name: etf.name,
            market: etf.market,
            type: "etf",
            category: etf.category,
            trackingIndex: etf.trackingIndex,
            scale: etf.scale,
          },
        });
        if (result.createdAt.getTime() === result.updatedAt.getTime()) {
          inserted++;
        } else {
          updated++;
        }
      } catch (error) {
        console.error(`存储ETF失败: ${etf.code}`, error);
      }
    }

    console.log(`✅ 存入数据库: 新增 ${inserted} 条, 更新 ${updated} 条`);

    return ok({
      total: etfs.length,
      inserted,
      updated,
      etfs,
    });
  }

  /**
   * 完整同步流程：ETF列表 + K线数据
   */
  async syncAllETFs(
    options: ETFFilterOptions & {
      syncKline?: boolean;
      klineBeginDate?: string;
    } = {},
  ): Promise<{
    etfListResult: { total: number; inserted: number; updated: number };
    klineResults: Array<{
      code: string;
      name: string;
      success: boolean;
      error?: string;
    }>;
  }> {
    const { syncKline = false, klineBeginDate, ...filterOptions } = options;

    const listResult = await this.syncETFList(filterOptions);
    if (!listResult.ok) {
      throw new Error(`ETF列表同步失败: ${listResult.error}`);
    }

    const etfs = listResult.value.etfs;
    const klineResults: Array<{
      code: string;
      name: string;
      success: boolean;
      error?: string;
    }> = [];

    if (syncKline) {
      console.log(`📊 开始同步 ${etfs.length} 只ETF的K线数据...`);

      for (const etf of etfs) {
        const result = await this.syncETF(etf.code, etf.name, {
          market: etf.market,
          category: etf.category,
          incremental: true,
          beginDate: klineBeginDate,
        });

        if (result.ok) {
          klineResults.push({ code: etf.code, name: etf.name, success: true });
        } else {
          klineResults.push({
            code: etf.code,
            name: etf.name,
            success: false,
            error: result.error,
          });
        }
      }

      const successCount = klineResults.filter((r) => r.success).length;
      console.log(`✅ K线同步完成: ${successCount}/${etfs.length} 成功`);
    }

    return {
      etfListResult: {
        total: listResult.value.total,
        inserted: listResult.value.inserted,
        updated: listResult.value.updated,
      },
      klineResults,
    };
  }

  async disconnect(): Promise<void> {
    await this.prisma.$disconnect();
  }
}
```

## 6. 阶段收益计算器

```typescript
// packages/data/src/period-return-calculator.ts

import { PrismaClient, Kline } from "@prisma/client";

export type PeriodType = "weekly" | "monthly" | "quarterly" | "yearly" | "ytd";

export interface PeriodReturnResult {
  code: string;
  periodType: PeriodType;
  periodKey: string;
  startDate: number;
  endDate: number;
  startPrice: number;
  endPrice: number;
  returnRate: number;
  returnAmount: number;
  tradeDays: number;
}

/**
 * 阶段收益计算器 - 全量重算并覆盖存储
 */
export class PeriodReturnCalculator {
  private prisma: PrismaClient;

  constructor(prisma?: PrismaClient) {
    this.prisma = prisma || new PrismaClient();
  }

  /**
   * 计算并保存所有阶段收益（全量覆盖）
   */
  async calculateAndSaveAll(code: string): Promise<{
    weekly: number;
    monthly: number;
    quarterly: number;
    yearly: number;
    ytd: number;
  }> {
    const klines = await this.prisma.kline.findMany({
      where: { code },
      orderBy: { date: "asc" },
    });

    if (klines.length === 0) {
      return { weekly: 0, monthly: 0, quarterly: 0, yearly: 0, ytd: 0 };
    }

    // 先删除旧数据
    await this.prisma.periodReturn.deleteMany({
      where: { code },
    });

    const allPeriods: PeriodReturnResult[] = [];

    // 计算各周期收益
    allPeriods.push(...this.calculatePeriodReturns(klines, "weekly"));
    allPeriods.push(...this.calculatePeriodReturns(klines, "monthly"));
    allPeriods.push(...this.calculatePeriodReturns(klines, "quarterly"));
    allPeriods.push(...this.calculatePeriodReturns(klines, "yearly"));

    const ytd = this.calculateYTD(klines);
    if (ytd) allPeriods.push(ytd);

    // 批量插入
    if (allPeriods.length > 0) {
      await this.prisma.periodReturn.createMany({
        data: allPeriods.map((p) => ({
          code: p.code,
          periodType: p.periodType,
          periodKey: p.periodKey,
          startDate: p.startDate,
          endDate: p.endDate,
          startPrice: p.startPrice,
          endPrice: p.endPrice,
          returnRate: p.returnRate,
          returnAmount: p.returnAmount,
          tradeDays: p.tradeDays,
        })),
      });
    }

    return {
      weekly: allPeriods.filter((p) => p.periodType === "weekly").length,
      monthly: allPeriods.filter((p) => p.periodType === "monthly").length,
      quarterly: allPeriods.filter((p) => p.periodType === "quarterly").length,
      yearly: allPeriods.filter((p) => p.periodType === "yearly").length,
      ytd: ytd ? 1 : 0,
    };
  }

  private calculatePeriodReturns(
    klines: Kline[],
    periodType: PeriodType,
  ): PeriodReturnResult[] {
    const groups = this.groupByPeriod(klines, periodType);
    const results: PeriodReturnResult[] = [];

    for (const [key, items] of groups) {
      if (items.length < 2) continue;
      const sorted = items.sort((a, b) => a.date - b.date);
      const first = sorted[0];
      const last = sorted[sorted.length - 1];

      results.push({
        code: first.code,
        periodType,
        periodKey: key,
        startDate: first.date,
        endDate: last.date,
        startPrice: first.open,
        endPrice: last.close,
        returnRate: (last.close - first.open) / first.open,
        returnAmount: last.close - first.open,
        tradeDays: sorted.length,
      });
    }

    return results;
  }

  private groupByPeriod(
    klines: Kline[],
    periodType: PeriodType,
  ): Map<string, Kline[]> {
    const groups = new Map<string, Kline[]>();

    for (const k of klines) {
      const date = new Date(k.date);
      let key: string;

      switch (periodType) {
        case "weekly": {
          const year = date.getFullYear();
          const week = this.getWeekNumber(date);
          key = `${year}-W${String(week).padStart(2, "0")}`;
          break;
        }
        case "monthly": {
          key = `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
          break;
        }
        case "quarterly": {
          const quarter = Math.floor(date.getMonth() / 3) + 1;
          key = `${date.getFullYear()}-Q${quarter}`;
          break;
        }
        case "yearly": {
          key = `${date.getFullYear()}`;
          break;
        }
        default:
          continue;
      }

      if (!groups.has(key)) {
        groups.set(key, []);
      }
      groups.get(key)!.push(k);
    }

    return groups;
  }

  private calculateYTD(klines: Kline[]): PeriodReturnResult | null {
    if (klines.length < 2) return null;

    const year = new Date(klines[0].date).getFullYear();
    const yearKlines = klines.filter((k) => {
      const d = new Date(k.date);
      return d.getFullYear() === year;
    });

    if (yearKlines.length < 2) return null;

    const first = yearKlines[0];
    const last = yearKlines[yearKlines.length - 1];

    return {
      code: first.code,
      periodType: "ytd",
      periodKey: `${year}-ytd`,
      startDate: first.date,
      endDate: last.date,
      startPrice: first.open,
      endPrice: last.close,
      returnRate: (last.close - first.open) / first.open,
      returnAmount: last.close - first.open,
      tradeDays: yearKlines.length,
    };
  }

  private getWeekNumber(date: Date): number {
    const d = new Date(date);
    d.setHours(0, 0, 0, 0);
    d.setDate(d.getDate() + 3 - ((d.getDay() + 6) % 7));
    const week1 = new Date(d.getFullYear(), 0, 4);
    return (
      1 +
      Math.round(
        ((d.getTime() - week1.getTime()) / 86400000 -
          3 +
          ((week1.getDay() + 6) % 7)) /
          7,
      )
    );
  }

  async disconnect(): Promise<void> {
    await this.prisma.$disconnect();
  }
}
```
