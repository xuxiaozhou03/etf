# 03-数据服务层

## 📖 文档概述

本文档详细描述量化回测系统的**数据服务层**，涵盖日K线读取、阶段收益查询、ETF排行榜服务等核心功能。

**目标读者：** 后端/数据开发工程师

## 1. 概述

### 1.1 数据服务层职责

| 职责             | 说明                                     |
| :--------------- | :--------------------------------------- |
| **日K线读取**    | 从Kline表读取数据，转换为DailyCandle格式 |
| **阶段收益查询** | 从PeriodReturn表读取周/月/季/年/YTD收益  |
| **ETF排行榜**    | 单周期排行、多周期综合排行、收益对比     |
| **工具方法**     | 获取股票列表、日期范围、数据验证         |

### 1.2 数据流

```
┌─────────────────────────────────────────────────────────────────┐
│                    数据服务层                                   │
│                                                               │
│  ┌─────────────────┐    ┌─────────────────┐                   │
│  │   loadCandles   │    │ getPeriodReturns│                   │
│  │   读取日K线      │    │  读取阶段收益    │                   │
│  └────────┬────────┘    └────────┬────────┘                   │
│           │                      │                             │
│           ▼                      ▼                             │
│  ┌─────────────────────────────────────────────┐               │
│  │            ETF排行榜服务                     │               │
│  │  getRanking / getMultiPeriodRanking         │               │
│  │  getComparison                              │               │
│  └─────────────────────────────────────────────┘               │
└─────────────────────────────────────────────────────────────────┘
```

## 3. 数据服务实现

```typescript
// packages/data/src/data-service.ts

import { PrismaClient } from "@prisma/client";
import { DailyCandle, Result, ok, err } from "@quant-backtest/core";
import { PeriodReturn, PeriodType, RankingQuery, RankingItem } from "./types";

/**
 * 数据加载配置
 */
export interface DataLoadConfig {
  code: string;
  startDate?: number;
  endDate?: number;
  limit?: number;
}

/**
 * 数据验证结果
 */
export interface ValidationResult {
  isValid: boolean;
  errors: string[];
  warnings: string[];
  totalRows: number;
}

/**
 * 数据服务
 * 从SQLite读取数据并转换为系统内部格式
 */
export class DataService {
  private prisma: PrismaClient;

  constructor(prisma?: PrismaClient) {
    this.prisma = prisma || new PrismaClient();
  }

  // ============================================================
  // 1. 日K线读取
  // ============================================================

  /**
   * 加载单只股票的日K线数据
   */
  async loadCandles(config: DataLoadConfig): Promise<DailyCandle[]> {
    const { code, startDate, endDate, limit } = config;

    const where: any = { code };
    if (startDate) where.date = { ...where.date, gte: startDate };
    if (endDate) where.date = { ...where.date, lte: endDate };

    const klines = await this.prisma.kline.findMany({
      where,
      orderBy: { date: "asc" },
      take: limit,
    });

    return klines.map((k, index) => ({
      symbol: k.code,
      date: k.date,
      prevClose: index > 0 ? klines[index - 1].close : k.open,
      open: k.open,
      high: k.high,
      low: k.low,
      close: k.close,
      volume: k.volume,
      amount: k.amount,
      factor: 1.0,
    }));
  }

  /**
   * 加载多只股票的K线数据
   */
  async loadMultipleCandles(
    codes: string[],
    startDate?: number,
    endDate?: number,
  ): Promise<Map<string, DailyCandle[]>> {
    const result = new Map<string, DailyCandle[]>();
    for (const code of codes) {
      const candles = await this.loadCandles({ code, startDate, endDate });
      result.set(code, candles);
    }
    return result;
  }

  /**
   * 验证数据完整性
   */
  validate(candles: DailyCandle[]): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    for (let i = 0; i < candles.length; i++) {
      const c = candles[i];

      if (c.symbol.length === 0) {
        errors.push(`第${i}行: 股票代码为空`);
      }
      if (c.close <= 0) {
        errors.push(`第${i}行: 收盘价无效 (${c.close})`);
      }
      if (c.high < c.low) {
        errors.push(`第${i}行: 最高价(${c.high}) < 最低价(${c.low})`);
      }
      if (i > 0 && candles[i - 1].date >= c.date) {
        errors.push(
          `第${i}行: 日期未升序 (${candles[i - 1].date} -> ${c.date})`,
        );
      }
    }

    return {
      isValid: errors.length === 0,
      errors,
      warnings,
      totalRows: candles.length,
    };
  }

  // ============================================================
  // 2. 阶段收益查询
  // ============================================================

  /**
   * 获取阶段收益
   */
  async getPeriodReturns(
    code: string,
    periodType?: PeriodType,
    periodKey?: string,
  ): Promise<PeriodReturn[]> {
    const where: any = { code };
    if (periodType) where.periodType = periodType;
    if (periodKey) where.periodKey = periodKey;

    const records = await this.prisma.periodReturn.findMany({
      where,
      orderBy: { startDate: "asc" },
    });

    return records.map((r) => ({
      code: r.code,
      periodType: r.periodType as PeriodType,
      periodKey: r.periodKey,
      startDate: r.startDate,
      endDate: r.endDate,
      startPrice: r.startPrice,
      endPrice: r.endPrice,
      returnRate: r.returnRate,
      returnAmount: r.returnAmount,
      tradeDays: r.tradeDays,
    }));
  }

  /**
   * 获取月度收益
   */
  async getMonthlyReturns(
    code: string,
    startDate?: number,
    endDate?: number,
  ): Promise<PeriodReturn[]> {
    return this.getPeriodReturns(code, "monthly");
  }

  /**
   * 获取年度收益
   */
  async getYearlyReturns(
    code: string,
    startDate?: number,
    endDate?: number,
  ): Promise<PeriodReturn[]> {
    return this.getPeriodReturns(code, "yearly");
  }

  /**
   * 获取完整收益汇总
   */
  async getReturnSummary(
    code: string,
    startDate?: number,
    endDate?: number,
  ): Promise<{
    cumulative: {
      totalReturn: number;
      annualizedReturn: number;
      holdingDays: number;
    };
    monthly: PeriodReturn[];
    yearly: PeriodReturn[];
    monthlyStats: {
      winRate: number;
      maxReturn: number;
      minReturn: number;
      avgReturn: number;
    };
  }> {
    const candles = await this.loadCandles({ code, startDate, endDate });

    if (candles.length < 2) {
      return {
        cumulative: { totalReturn: 0, annualizedReturn: 0, holdingDays: 0 },
        monthly: [],
        yearly: [],
        monthlyStats: { winRate: 0, maxReturn: 0, minReturn: 0, avgReturn: 0 },
      };
    }

    // 累计收益
    const first = candles[0];
    const last = candles[candles.length - 1];
    const totalReturn = (last.close - first.open) / first.open;
    const holdingDays = candles.length;
    const annualizedReturn = Math.pow(1 + totalReturn, 365 / holdingDays) - 1;

    // 月度收益
    const monthly = await this.getMonthlyReturns(code);
    const monthlyReturns = monthly.map((m) => m.returnRate);
    const positive = monthlyReturns.filter((r) => r > 0);

    return {
      cumulative: { totalReturn, annualizedReturn, holdingDays },
      monthly,
      yearly: await this.getYearlyReturns(code),
      monthlyStats: {
        winRate:
          monthlyReturns.length > 0
            ? positive.length / monthlyReturns.length
            : 0,
        maxReturn: monthlyReturns.length > 0 ? Math.max(...monthlyReturns) : 0,
        minReturn: monthlyReturns.length > 0 ? Math.min(...monthlyReturns) : 0,
        avgReturn:
          monthlyReturns.length > 0
            ? monthlyReturns.reduce((a, b) => a + b, 0) / monthlyReturns.length
            : 0,
      },
    };
  }

  // ============================================================
  // 3. ETF排行榜服务
  // ============================================================

  /**
   * 获取ETF排行榜
   */
  async getRanking(
    query: RankingQuery,
  ): Promise<Result<RankingItem[], string>> {
    try {
      const {
        periodType,
        periodKey,
        sortBy,
        order,
        limit = 100,
        minTradeDays = 0,
        category,
        market,
      } = query;

      // 构建查询条件
      const where: any = { periodType };
      if (periodKey) where.periodKey = periodKey;
      if (minTradeDays > 0) where.tradeDays = { gte: minTradeDays };

      // 关联Security表筛选
      const securityWhere: any = { isActive: true };
      if (category) securityWhere.category = category;
      if (market) securityWhere.market = market;

      const records = await this.prisma.periodReturn.findMany({
        where,
        include: {
          security: true,
        },
        orderBy: {
          returnRate: order,
        },
        take: limit,
      });

      const items: RankingItem[] = records.map((r) => ({
        code: r.code,
        name: r.security?.name || r.code,
        market: r.security?.market || "",
        category: r.security?.category || "",
        periodKey: r.periodKey,
        returnRate: r.returnRate,
        returnAmount: r.returnAmount,
        tradeDays: r.tradeDays,
      }));

      return ok(items);
    } catch (error) {
      return err(error instanceof Error ? error.message : String(error));
    }
  }

  /**
   * 获取多周期综合排行
   */
  async getMultiPeriodRanking(
    periodKey: string,
    options?: {
      limit?: number;
      minTradeDays?: number;
      category?: string;
    },
  ): Promise<RankingItem[]> {
    const { limit = 50, minTradeDays = 0, category } = options || {};

    // 获取所有活跃的ETF
    const securities = await this.prisma.security.findMany({
      where: {
        type: "etf",
        ...(category && { category }),
      },
    });

    if (securities.length === 0) return [];

    const codes = securities.map((s) => s.code);

    // 查询多个周期收益
    const periodReturns = await this.prisma.periodReturn.findMany({
      where: {
        code: { in: codes },
        periodKey: periodKey,
        periodType: { in: ["weekly", "monthly", "quarterly", "yearly", "ytd"] },
        ...(minTradeDays > 0 && { tradeDays: { gte: minTradeDays } }),
      },
    });

    // 按code分组
    const grouped = new Map<string, any>();
    for (const pr of periodReturns) {
      if (!grouped.has(pr.code)) {
        grouped.set(pr.code, { code: pr.code });
      }
      const item = grouped.get(pr.code);
      item[`${pr.periodType}Return`] = pr.returnRate;
      item[`${pr.periodType}Days`] = pr.tradeDays;
    }

    // 合并Security信息
    const result: RankingItem[] = [];
    for (const [code, data] of grouped) {
      const sec = securities.find((s) => s.code === code);
      result.push({
        code,
        name: sec?.name || code,
        market: sec?.market || "",
        category: sec?.category || "",
        periodKey,
        returnRate: data.monthlyReturn || 0,
        returnAmount: 0,
        tradeDays: data.monthlyDays || 0,
        weeklyReturn: data.weeklyReturn,
        monthlyReturn: data.monthlyReturn,
        quarterlyReturn: data.quarterlyReturn,
        yearlyReturn: data.yearlyReturn,
        ytdReturn: data.ytdReturn,
      });
    }

    // 按月度收益排序
    return result
      .sort((a, b) => (b.monthlyReturn || 0) - (a.monthlyReturn || 0))
      .slice(0, limit);
  }

  /**
   * 获取多个ETF的收益对比
   */
  async getComparison(
    codes: string[],
    periodKeys: string[],
  ): Promise<Map<string, Map<string, number>>> {
    const results = await this.prisma.periodReturn.findMany({
      where: {
        code: { in: codes },
        periodKey: { in: periodKeys },
        periodType: "monthly",
      },
    });

    const map = new Map<string, Map<string, number>>();
    for (const r of results) {
      if (!map.has(r.code)) {
        map.set(r.code, new Map());
      }
      map.get(r.code)!.set(r.periodKey, r.returnRate);
    }

    return map;
  }

  // ============================================================
  // 4. 工具方法
  // ============================================================

  /**
   * 获取可用股票列表
   */
  async getAvailableSymbols(): Promise<string[]> {
    const results = await this.prisma.kline.groupBy({
      by: ["code"],
    });
    return results.map((r) => r.code);
  }

  /**
   * 获取数据日期范围
   */
  async getDateRange(code: string): Promise<{ start: number; end: number }> {
    const result = await this.prisma.kline.aggregate({
      where: { code },
      _min: { date: true },
      _max: { date: true },
    });

    return {
      start: result._min.date || 0,
      end: result._max.date || 0,
    };
  }

  /**
   * 获取所有ETF列表（含分类）
   */
  async getETFList(options?: { category?: string; market?: string }): Promise<
    Array<{
      code: string;
      name: string;
      market: string;
      category: string;
      trackingIndex: string;
      scale: number;
    }>
  > {
    const where: any = { type: "etf" };
    if (options?.category) where.category = options.category;
    if (options?.market) where.market = options.market;

    const records = await this.prisma.security.findMany({
      where,
      orderBy: { scale: "desc" },
    });

    return records.map((r) => ({
      code: r.code,
      name: r.name,
      market: r.market,
      category: r.category || "",
      trackingIndex: r.trackingIndex || "",
      scale: r.scale || 0,
    }));
  }

  async disconnect(): Promise<void> {
    await this.prisma.$disconnect();
  }
}
```

## 5. 使用示例

### 5.1 查询日K线

```typescript
import { DataService } from "@quant-backtest/data";

const dataService = new DataService();

// 加载单只ETF的日K线
const candles = await dataService.loadCandles({
  code: "510300.SH",
  startDate: 20240101,
  endDate: 20241231,
});

console.log(`加载了 ${candles.length} 条数据`);
console.log(`首日: ${candles[0].date}, 收盘: ${candles[0].close}`);
console.log(
  `末日: ${candles[candles.length - 1].date}, 收盘: ${candles[candles.length - 1].close}`,
);
```

### 5.2 查询阶段收益

```typescript
import { DataService } from "@quant-backtest/data";

const dataService = new DataService();

// 获取月度收益
const monthly = await dataService.getMonthlyReturns("510300.SH");
console.log(`月度收益: ${monthly.length} 条`);
for (const m of monthly.slice(-6)) {
  console.log(`  ${m.periodKey}: ${(m.returnRate * 100).toFixed(2)}%`);
}

// 获取完整汇总
const summary = await dataService.getReturnSummary(
  "510300.SH",
  20240101,
  20241231,
);
console.log(`
📊 收益汇总:
  累计收益: ${(summary.cumulative.totalReturn * 100).toFixed(2)}%
  年化收益: ${(summary.cumulative.annualizedReturn * 100).toFixed(2)}%
  持有天数: ${summary.cumulative.holdingDays} 天
  月度胜率: ${(summary.monthlyStats.winRate * 100).toFixed(2)}%
`);
```

### 5.3 ETF排行榜

```typescript
import { DataService } from "@quant-backtest/data";

const dataService = new DataService();

// 月度收益排行
const result = await dataService.getRanking({
  periodType: "monthly",
  periodKey: "2024-12",
  sortBy: "monthlyReturn",
  order: "desc",
  limit: 10,
  minTradeDays: 10,
});

if (result.ok) {
  console.log("📊 2024-12 月度收益排行:");
  for (const item of result.value) {
    console.log(
      `  ${item.code} ${item.name}: ${(item.returnRate * 100).toFixed(2)}% (${item.tradeDays}天)`,
    );
  }
}

// 多周期综合排行
const multiRank = await dataService.getMultiPeriodRanking("2024", {
  limit: 10,
  minTradeDays: 100,
});

console.log("\n📊 2024年多周期综合表现:");
for (const item of multiRank) {
  console.log(
    `  ${item.code} ${item.name}: ` +
      `月${((item.monthlyReturn || 0) * 100).toFixed(2)}% | ` +
      `年${((item.yearlyReturn || 0) * 100).toFixed(2)}%`,
  );
}
```
