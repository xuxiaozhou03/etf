# 04-策略层

## 📖 文档概述

本文档详细描述量化回测系统的**策略层**，涵盖策略接口定义、基础策略类（内置风控）、参数配置、组合策略等核心功能。

**目标读者：** 策略开发工程师

## 1. 概述

### 1.1 策略层职责

| 职责         | 说明                                     |
| :----------- | :--------------------------------------- |
| **交易决策** | 接收市场数据，输出买卖信号               |
| **状态维护** | 维护策略内部状态（持仓标记、指标缓存等） |
| **风控能力** | 止损、止盈、仓位控制、日亏损限制         |
| **参数配置** | 支持策略参数定义和优化                   |
| **组合策略** | 支持多股票同时回测                       |

### 1.2 策略生命周期

```
┌─────────────────────────────────────────────────────────────────┐
│                     策略生命周期                                │
│                                                               │
```

## 3. 策略接口

```typescript
// packages/strategy/src/strategy.ts

import {
  DailyCandle,
  AdviceContext,
  Signal,
  FilledOrder,
  RiskAction,
} from "@quant-backtest/core";

/**
 * 策略接口
 * 所有策略必须实现此接口
 */
export interface Strategy {
  /** 策略名称 */
  readonly name?: string;

  /**
   * 策略初始化
   * 在回测开始时调用一次，用于设置初始状态
   */
  init?(): void;

  /**
   * 核心决策方法
   * 在每个交易日调用，是策略的核心
   * @param candle - 当日的日K线数据
   * @param context - 决策上下文
   * @returns 交易信号，或 null（表示不操作）
   */
  next(candle: DailyCandle, context: AdviceContext): Signal | null;

  /**
   * 每日风险检查
   * 在每个交易日开始前调用，用于检查是否需要强制平仓
   * @param context - 决策上下文
   * @returns 风控动作，或 null（表示无需操作）
   */
  onRiskCheck?(context: AdviceContext): RiskAction | null;

  /**
   * 持仓监控
   * 每日收盘后调用，检查是否需要调整持仓
   * @param position - 当前持仓
   * @param context - 决策上下文
   * @returns 交易信号，或 null（表示无需操作）
   */
  onPositionMonitor?(position: any, context: AdviceContext): Signal | null;

  /**
   * 订单成交回调
   * 在每次交易执行后调用，用于更新策略内部状态
   */
  onOrderFilled?(order: FilledOrder): void;

  /**
   * 策略清理
   * 在回测结束时调用，用于释放资源
   */
  cleanup?(): void;
}

/**
 * 策略构造函数类型
 */
export type StrategyConstructor = new (...args: any[]) => Strategy;
```

## 4. 基础策略类（内置风控）

```typescript
// packages/strategy/src/base-strategy.ts

import {
  DailyCandle,
  AdviceContext,
  Signal,
  FilledOrder,
  RiskAction,
} from "@quant-backtest/core";
import { Strategy } from "./strategy";

/**
 * 基础策略类 - 内置风控能力
 * 策略可以继承此类，获得默认的止损止盈能力
 */
export abstract class BaseStrategy implements Strategy {
  readonly name = "基础策略";

  /** 止损比例（如 0.05 表示跌5%止损） */
  protected stopLossRate?: number;
  /** 止盈比例（如 0.10 表示涨10%止盈） */
  protected takeProfitRate?: number;
  /** 单只股票最大仓位比例（占总资产比例） */
  protected maxPositionRate?: number;
  /** 每日最大亏损比例 */
  protected dailyLossLimitRate?: number;

  /** 是否已记录今日亏损 */
  private todayHasLoss = false;

  constructor(config?: {
    stopLossRate?: number;
    takeProfitRate?: number;
    maxPositionRate?: number;
    dailyLossLimitRate?: number;
  }) {
    this.stopLossRate = config?.stopLossRate;
    this.takeProfitRate = config?.takeProfitRate;
    this.maxPositionRate = config?.maxPositionRate;
    this.dailyLossLimitRate = config?.dailyLossLimitRate;
  }

  /**
   * 策略子类必须实现的核心决策方法
   */
  abstract next(candle: DailyCandle, context: AdviceContext): Signal | null;

  /**
   * 每日风险检查
   * 检查是否触发止损、止盈或每日亏损限制
   */
  onRiskCheck(context: AdviceContext): RiskAction | null {
    const symbol = context.symbol;
    const position = context.positions[symbol];

    // 没有持仓，无需检查
    if (!position || position.quantity === 0) {
      return null;
    }

    const currentPrice =
      context.historyCandles[context.historyCandles.length - 1]?.close;
    if (!currentPrice) return null;

    // 1. 止损检查
    if (this.stopLossRate) {
      const lossRate = (currentPrice - position.costPrice) / position.costPrice;
      if (lossRate <= -this.stopLossRate) {
        return {
          type: "stop_loss",
          signal: {
            action: "sell",
            quantity: position.quantity,
            price: currentPrice,
            reason: `止损触发: 亏损${(lossRate * 100).toFixed(2)}%`,
          },
          reason: `止损触发: 亏损${(lossRate * 100).toFixed(2)}%`,
        };
      }
    }

    // 2. 止盈检查
    if (this.takeProfitRate) {
      const profitRate =
        (currentPrice - position.costPrice) / position.costPrice;
      if (profitRate >= this.takeProfitRate) {
        return {
          type: "take_profit",
          signal: {
            action: "sell",
            quantity: position.quantity,
            price: currentPrice,
            reason: `止盈触发: 盈利${(profitRate * 100).toFixed(2)}%`,
          },
          reason: `止盈触发: 盈利${(profitRate * 100).toFixed(2)}%`,
        };
      }
    }

    // 3. 每日亏损限制检查
    if (this.dailyLossLimitRate) {
      const dailyReturn = this.calculateDailyReturn(context);
      if (dailyReturn <= -this.dailyLossLimitRate && !this.todayHasLoss) {
        this.todayHasLoss = true;
        return {
          type: "daily_loss_limit",
          signal: null,
          reason: `日亏损限制触发: 今日亏损${(dailyReturn * 100).toFixed(2)}%`,
        };
      }
    }

    return null;
  }

  /**
   * 持仓监控
   * 检查是否需要调整持仓（如减仓）
   */
  onPositionMonitor(position: any, context: AdviceContext): Signal | null {
    if (this.maxPositionRate) {
      const totalAssets = context.balances.total;
      const currentPrice =
        context.historyCandles[context.historyCandles.length - 1]?.close;
      if (!currentPrice) return null;

      const currentPositionValue = position.quantity * currentPrice;
      const currentRate = currentPositionValue / totalAssets;

      if (currentRate > this.maxPositionRate) {
        const targetQuantity = Math.floor(
          (this.maxPositionRate * totalAssets) / currentPrice,
        );
        const reduceQuantity = position.quantity - targetQuantity;
        if (reduceQuantity > 0) {
          return {
            action: "sell",
            quantity: reduceQuantity,
            price: currentPrice,
            reason: `仓位控制: 当前占比${(currentRate * 100).toFixed(2)}%，目标${(this.maxPositionRate * 100).toFixed(2)}%`,
          };
        }
      }
    }

    return null;
  }

  /**
   * 计算当日收益率
   */
  private calculateDailyReturn(context: AdviceContext): number {
    const snapshots = context.historyCandles;
    if (snapshots.length < 2) return 0;
    const prevClose = snapshots[snapshots.length - 2].close;
    const currClose = snapshots[snapshots.length - 1].close;
    return (currClose - prevClose) / prevClose;
  }

  /**
   * 重置每日状态（由回测引擎在每日开始前调用）
   */
  resetDailyState(): void {
    this.todayHasLoss = false;
  }

  /**
   * 策略清理
   */
  cleanup?(): void;
}
```

## 7. 参数优化工具

```typescript
// packages/strategy/src/parameter-optimizer.ts

import { ConfigurableStrategy, StrategyParams } from "./configurable-strategy";

/**
 * 参数优化工具
 * 支持网格搜索和随机搜索
 */
export class ParameterOptimizer {
  /**
   * 网格搜索
   * 在参数范围内遍历所有组合，找到最优参数
   */
  async gridSearch<T extends ConfigurableStrategy>(
    strategyClass: new (...args: any[]) => T,
    paramRanges: {
      [key: string]: { min: number; max: number; step: number };
    },
    evaluate: (params: StrategyParams) => Promise<number>,
    options?: {
      maxCombinations?: number;
      metric?: "max" | "min";
    },
  ): Promise<{
    bestParams: StrategyParams;
    bestScore: number;
    allResults: Array<{ params: StrategyParams; score: number }>;
  }> {
    const metric = options?.metric || "max";
    const maxCombinations = options?.maxCombinations || 10000;

    // 生成所有参数组合
    const combinations = this.generateParamCombinations(paramRanges);

    if (combinations.length > maxCombinations) {
      console.warn(
        `参数组合数 ${combinations.length} 超过限制 ${maxCombinations}，将随机采样`,
      );
      return this.randomSearch(strategyClass, paramRanges, evaluate, {
        maxIterations: maxCombinations,
        metric,
      });
    }

    // 评估所有组合
    const results = await this.evaluateAll(combinations, evaluate);

    // 选择最优
    const best = results.reduce((best, current) => {
      if (metric === "max") {
        return current.score > best.score ? current : best;
      } else {
        return current.score < best.score ? current : best;
      }
    });

    return {
      bestParams: best.params,
      bestScore: best.score,
      allResults: results,
    };
  }

  /**
   * 随机搜索
   * 在参数范围内随机采样
   */
  async randomSearch<T extends ConfigurableStrategy>(
    strategyClass: new (...args: any[]) => T,
    paramRanges: {
      [key: string]: { min: number; max: number; step: number };
    },
    evaluate: (params: StrategyParams) => Promise<number>,
    options?: {
      maxIterations?: number;
      metric?: "max" | "min";
    },
  ): Promise<{
    bestParams: StrategyParams;
    bestScore: number;
    allResults: Array<{ params: StrategyParams; score: number }>;
  }> {
    const maxIterations = options?.maxIterations || 100;
    const metric = options?.metric || "max";

    const results: Array<{ params: StrategyParams; score: number }> = [];
    const keys = Object.keys(paramRanges);

    for (let i = 0; i < maxIterations; i++) {
      const params: StrategyParams = {};
      for (const key of keys) {
        const range = paramRanges[key];
        const steps = Math.floor((range.max - range.min) / range.step);
        const randomStep = Math.floor(Math.random() * (steps + 1));
        params[key] =
          Math.round((range.min + randomStep * range.step) * 100) / 100;
      }

      const score = await evaluate(params);
      results.push({ params, score });
    }

    const best = results.reduce((best, current) => {
      if (metric === "max") {
        return current.score > best.score ? current : best;
      } else {
        return current.score < best.score ? current : best;
      }
    });

    return {
      bestParams: best.params,
      bestScore: best.score,
      allResults: results,
    };
  }

  private generateParamCombinations(paramRanges: {
    [key: string]: { min: number; max: number; step: number };
  }): StrategyParams[] {
    const keys = Object.keys(paramRanges);
    const ranges = keys.map((key) => {
      const r = paramRanges[key];
      const values: number[] = [];
      for (let v = r.min; v <= r.max; v += r.step) {
        values.push(Math.round(v * 100) / 100);
      }
      return { key, values };
    });

    const combinations: StrategyParams[] = [];

    const generate = (index: number, current: StrategyParams) => {
      if (index === ranges.length) {
        combinations.push({ ...current });
        return;
      }

      const { key, values } = ranges[index];
      for (const value of values) {
        current[key] = value;
        generate(index + 1, current);
      }
    };

    generate(0, {});
    return combinations;
  }

  private async evaluateAll(
    combinations: StrategyParams[],
    evaluate: (params: StrategyParams) => Promise<number>,
  ): Promise<Array<{ params: StrategyParams; score: number }>> {
    const results: Array<{ params: StrategyParams; score: number }> = [];

    for (const params of combinations) {
      const score = await evaluate(params);
      results.push({ params, score });
    }

    return results;
  }
}
```

## 8. 策略示例

### 8.1 双均线策略（无风控）

```typescript
// examples/strategies/double-ma.ts

import {
  DailyCandle,
  AdviceContext,
  Signal,
  Strategy,
} from "@quant-backtest/core";
import { calculateSMA } from "../indicators"; // 第三方技术指标库

/**
 * 双均线策略
 * 金叉买入，死叉卖出
 */
export class DoubleMAStrategy implements Strategy {
  readonly name = "双均线策略";
  private shortPeriod: number;
  private longPeriod: number;
  private hasPosition = false;

  constructor(shortPeriod: number = 5, longPeriod: number = 20) {
    this.shortPeriod = shortPeriod;
    this.longPeriod = longPeriod;
  }

  init(): void {
    console.log(`${this.name} 初始化完成`);
  }

  next(candle: DailyCandle, context: AdviceContext): Signal | null {
    const prices = context.historyCandles.map((c) => c.close);
    const shortMA = calculateSMA(prices, this.shortPeriod);
    const longMA = calculateSMA(prices, this.longPeriod);

    const lastShort = shortMA[shortMA.length - 1];
    const lastLong = longMA[longMA.length - 1];
    const prevShort = shortMA[shortMA.length - 2];
    const prevLong = longMA[longMA.length - 2];

    if (lastShort === undefined || lastLong === undefined) return null;

    // 金叉买入
    if (prevShort <= prevLong && lastShort > lastLong && !this.hasPosition) {
      const shares = Math.floor((context.balances.free * 0.8) / candle.close);
      if (shares > 0) {
        this.hasPosition = true;
        return {
          action: "buy",
          quantity: shares,
          price: candle.close,
          reason: `金叉: MA${this.shortPeriod}(${lastShort.toFixed(2)}) > MA${this.longPeriod}(${lastLong.toFixed(2)})`,
        };
      }
    }

    // 死叉卖出
    if (prevShort >= prevLong && lastShort < lastLong && this.hasPosition) {
      const position = context.positions[context.symbol];
      if (position) {
        this.hasPosition = false;
        return {
          action: "sell",
          quantity: position.quantity,
          price: candle.close,
          reason: `死叉: MA${this.shortPeriod}(${lastShort.toFixed(2)}) < MA${this.longPeriod}(${lastLong.toFixed(2)})`,
        };
      }
    }

    return null;
  }

  onOrderFilled(order: FilledOrder): void {
    console.log(
      `[${order.tradeDate}] ${order.side} ${order.quantity}股 @ ${order.price}`,
    );
  }

  cleanup(): void {
    console.log(`${this.name} 回测结束`);
  }
}
```

### 8.2 带风控的双均线策略

```typescript
// examples/strategies/safe-double-ma.ts

import { DailyCandle, AdviceContext, Signal } from "@quant-backtest/core";
import { BaseStrategy } from "@quant-backtest/strategy";
import { calculateSMA } from "../indicators";

/**
 * 带风控的双均线策略
 * 继承 BaseStrategy 获得自动止损止盈能力
 */
export class SafeDoubleMAStrategy extends BaseStrategy {
  readonly name = "带风控双均线策略";
  private shortPeriod: number;
  private longPeriod: number;
  private hasPosition = false;

  constructor(
    shortPeriod: number = 5,
    longPeriod: number = 20,
    config?: {
      stopLossRate?: number;
      takeProfitRate?: number;
      maxPositionRate?: number;
      dailyLossLimitRate?: number;
    },
  ) {
    super(config);
    this.shortPeriod = shortPeriod;
    this.longPeriod = longPeriod;
  }

  next(candle: DailyCandle, context: AdviceContext): Signal | null {
    const prices = context.historyCandles.map((c) => c.close);
    const shortMA = calculateSMA(prices, this.shortPeriod);
    const longMA = calculateSMA(prices, this.longPeriod);

    const lastShort = shortMA[shortMA.length - 1];
    const lastLong = longMA[longMA.length - 1];

    if (lastShort === undefined || lastLong === undefined) return null;

    // 金叉买入 - 使用仓位控制
    if (lastShort > lastLong && !this.hasPosition) {
      let shares = Math.floor((context.balances.free * 0.8) / candle.close);

      // 如果设置了最大仓位限制，进一步调整
      if (this.maxPositionRate) {
        const maxValue = context.balances.total * this.maxPositionRate;
        const maxShares = Math.floor(maxValue / candle.close);
        shares = Math.min(shares, maxShares);
      }

      if (shares > 0) {
        this.hasPosition = true;
        return {
          action: "buy",
          quantity: shares,
          price: candle.close,
          reason: `金叉买入: MA${this.shortPeriod} > MA${this.longPeriod}`,
        };
      }
    }

    // 死叉卖出
    if (lastShort < lastLong && this.hasPosition) {
      const position = context.positions[context.symbol];
      if (position) {
        this.hasPosition = false;
        return {
          action: "sell",
          quantity: position.quantity,
          price: candle.close,
          reason: `死叉卖出: MA${this.shortPeriod} < MA${this.longPeriod}`,
        };
      }
    }

    return null;
  }

  onOrderFilled(order: FilledOrder): void {
    console.log(
      `[${order.tradeDate.toISOString().split("T")[0]}] ` +
        `${order.side === "buy" ? "买入" : "卖出"} ${order.quantity}股 @ ${order.price}` +
        (order.reason ? ` (${order.reason})` : ""),
    );
  }
}
```

## 9. 策略开发规范

| 规范             | 说明                                            |
| :--------------- | :---------------------------------------------- |
| **无副作用**     | 策略不应修改外部状态，所有依赖通过 Context 注入 |
| **确定性**       | 相同输入应产生相同输出                          |
| **使用复权价格** | 所有价格计算使用 `DailyCandle.close`（已复权）  |
| **资金管理**     | 策略应检查 `balances.free` 确保资金充足         |
| **持仓检查**     | 卖出前检查 `positions` 确保有持仓               |
| **日志记录**     | 使用 `reason` 字段记录交易理由                  |
| **参数验证**     | 可配置策略应实现 `validateParams` 方法          |
