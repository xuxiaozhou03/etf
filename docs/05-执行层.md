# 05-执行层

## 📖 文档概述

本文档详细描述量化回测系统的**执行层**，涵盖模拟交易执行、账户状态管理、费用计算、交易明细记录和每日快照生成等核心功能。

**目标读者：** 交易/执行开发工程师

## 1. 概述

### 1.1 执行层职责

| 职责             | 说明                             |
| :--------------- | :------------------------------- |
| **模拟交易执行** | 接收策略信号，模拟买入/卖出成交  |
| **账户状态管理** | 维护现金余额、持仓数量、持仓成本 |
| **费用计算**     | 计算佣金、印花税、过户费         |
| **交易明细记录** | 记录每笔交易的完整信息           |
| **每日快照生成** | 记录每个交易日的完整状态         |
| **详细结果反馈** | 返回成功/失败及详细错误信息      |
| **策略回调**     | 通过 onOrderFilled 通知策略      |

## 4. 执行器实现

```typescript
// packages/execution/src/backtest-executor.ts

import {
  Account,
  Position,
  TradeRecord,
  DailySnapshot,
  SnapshotPosition,
  ExecutionResult,
  ExecutionResultData,
  ExecutionErrorReason,
  FilledOrder,
  FeeConfig,
  Signal,
  DailyCandle,
} from "@quant-backtest/core";

/**
 * 回测执行器
 * 模拟交易执行，维护账户状态
 */
export class BacktestExecutor {
  private account: Account;
  private trades: TradeRecord[] = [];
  private snapshots: DailySnapshot[] = [];
  private feeConfig: FeeConfig;
  private tradeCounter = 0;
  private executionErrors: Array<{
    timestamp: Date;
    symbol: string;
    error: string;
  }> = [];

  constructor(initialCash: number, feeConfig: FeeConfig) {
    this.account = {
      initialCash,
      cash: initialCash,
      positions: new Map(),
      totalFees: 0,
      totalRealizedProfit: 0,
    };
    this.feeConfig = feeConfig;
  }

  // ============================================================
  // 核心方法
  // ============================================================

  /**
   * 执行交易信号
   * @param signal - 交易信号
   * @param candle - 当日K线
   * @returns 详细的执行结果
   */
  executeSignal(signal: Signal, candle: DailyCandle): ExecutionResult {
    const { action, quantity, price, reason } = signal;

    // 1. 基本参数校验
    if (quantity <= 0) {
      return {
        success: false,
        reason: "INVALID_QUANTITY",
        message: `交易股数无效: ${quantity}，必须大于0`,
        suggestion: "请检查策略中的数量计算逻辑",
      };
    }

    if (price <= 0) {
      return {
        success: false,
        reason: "INVALID_PRICE",
        message: `交易价格无效: ${price}，必须大于0`,
        suggestion: "请检查是否使用了有效的复权价格",
      };
    }

    // 2. 执行交易
    if (action === "buy") {
      return this.executeBuy(
        candle.symbol,
        quantity,
        price,
        candle.date,
        reason,
      );
    } else if (action === "sell") {
      return this.executeSell(
        candle.symbol,
        quantity,
        price,
        candle.date,
        reason,
      );
    } else {
      return {
        success: false,
        reason: "UNKNOWN_ERROR",
        message: `未知的交易动作: ${action}`,
        suggestion: "请检查策略中的 action 字段",
      };
    }
  }

  /**
   * 生成每日快照
   * @param date - 快照日期
   * @param prices - 当前价格映射
   */
  snapshot(date: Date, prices: Map<string, number>): void {
    let marketValue = 0;
    const positions: SnapshotPosition[] = [];

    for (const [symbol, pos] of this.account.positions) {
      const currentPrice = prices.get(symbol) || 0;
      const mv = pos.quantity * currentPrice;
      marketValue += mv;
      positions.push({
        symbol,
        quantity: pos.quantity,
        costPrice: pos.costPrice,
        totalCost: pos.totalCost,
        currentPrice,
        marketValue: mv,
        profit: mv - pos.totalCost,
        profitRate: pos.totalCost > 0 ? mv / pos.totalCost - 1 : 0,
      });
    }

    const totalAssets = this.account.cash + marketValue;
    const unrealizedProfit =
      marketValue -
      Array.from(this.account.positions.values()).reduce(
        (sum, p) => sum + p.totalCost,
        0,
      );

    this.snapshots.push({
      date,
      cash: this.account.cash,
      marketValue,
      totalAssets,
      realizedProfit: this.account.totalRealizedProfit,
      unrealizedProfit,
      totalProfit: this.account.totalRealizedProfit + unrealizedProfit,
      returnRate:
        (totalAssets - this.account.initialCash) / this.account.initialCash,
      positions,
    });
  }

  /**
   * 获取执行结果
   */
  getResult(): ExecutionResultData {
    return {
      initialCash: this.account.initialCash,
      finalCash: this.account.cash,
      finalTotalAssets:
        this.snapshots[this.snapshots.length - 1]?.totalAssets ||
        this.account.cash,
      trades: this.trades,
      snapshots: this.snapshots,
      totalFees: this.account.totalFees,
      totalRealizedProfit: this.account.totalRealizedProfit,
      finalPositions: new Map(this.account.positions),
    };
  }

  /**
   * 获取当前账户状态
   */
  getAccount(): Account {
    return { ...this.account, positions: new Map(this.account.positions) };
  }

  /**
   * 获取执行错误列表
   */
  getExecutionErrors(): Array<{
    timestamp: Date;
    symbol: string;
    error: string;
  }> {
    return this.executionErrors;
  }

  // ============================================================
  // 私有方法
  // ============================================================

  /**
   * 执行买入
   */
  private executeBuy(
    symbol: string,
    quantity: number,
    price: number,
    date: number,
    reason?: string,
  ): ExecutionResult {
    const amount = quantity * price;
    const fee = this.calculateFee(amount, "buy");
    const totalCost = amount + fee;

    // 资金检查
    if (this.account.cash < totalCost) {
      return {
        success: false,
        reason: "INSUFFICIENT_CASH",
        message: `资金不足: 需要 ${totalCost.toFixed(2)}，可用 ${this.account.cash.toFixed(2)}`,
        suggestion: `建议减少买入数量，当前最多可买 ${Math.floor(this.account.cash / price)} 股`,
      };
    }

    // 更新现金
    this.account.cash -= totalCost;
    this.account.totalFees += fee;

    // 更新持仓
    const position = this.account.positions.get(symbol);
    let positionQuantityAfter: number;

    if (position) {
      const oldTotalCost = position.quantity * position.costPrice;
      const newTotalCost = oldTotalCost + amount;
      const newQuantity = position.quantity + quantity;
      position.costPrice = newTotalCost / newQuantity;
      position.quantity = newQuantity;
      position.totalCost = newTotalCost;
      positionQuantityAfter = newQuantity;
    } else {
      this.account.positions.set(symbol, {
        symbol,
        quantity,
        costPrice: price,
        totalCost: amount,
      });
      positionQuantityAfter = quantity;
    }

    // 记录交易
    const trade = this.createTradeRecord(
      symbol,
      "buy",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );
    this.trades.push(trade);

    // 生成成交订单
    const order = this.createFilledOrder(
      symbol,
      "buy",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );

    return {
      success: true,
      order,
      trade,
    };
  }

  /**
   * 执行卖出
   */
  private executeSell(
    symbol: string,
    quantity: number,
    price: number,
    date: number,
    reason?: string,
  ): ExecutionResult {
    const position = this.account.positions.get(symbol);

    // 持仓检查
    if (!position) {
      return {
        success: false,
        reason: "POSITION_NOT_FOUND",
        message: `未找到股票 ${symbol} 的持仓`,
        suggestion: "请检查是否持有该股票，或确认股票代码是否正确",
      };
    }

    if (position.quantity < quantity) {
      return {
        success: false,
        reason: "INSUFFICIENT_POSITION",
        message: `持仓不足: 需要 ${quantity}股，持有 ${position.quantity}股`,
        suggestion: `建议减少卖出数量，当前最多可卖 ${position.quantity} 股`,
      };
    }

    const amount = quantity * price;
    const fee = this.calculateFee(amount, "sell");
    const netRevenue = amount - fee;

    // 更新现金
    this.account.cash += netRevenue;
    this.account.totalFees += fee;

    // 计算已实现盈亏
    const costOfSold = quantity * position.costPrice;
    const realizedProfit = netRevenue - costOfSold;
    this.account.totalRealizedProfit += realizedProfit;

    // 更新持仓
    position.quantity -= quantity;
    let positionQuantityAfter: number;
    if (position.quantity === 0) {
      this.account.positions.delete(symbol);
      positionQuantityAfter = 0;
    } else {
      position.totalCost = position.quantity * position.costPrice;
      positionQuantityAfter = position.quantity;
    }

    // 记录交易
    const trade = this.createTradeRecord(
      symbol,
      "sell",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );
    this.trades.push(trade);

    // 生成成交订单
    const order = this.createFilledOrder(
      symbol,
      "sell",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );

    return {
      success: true,
      order,
      trade,
    };
  }

  /**
   * 计算手续费
   */
  private calculateFee(amount: number, side: "buy" | "sell"): number {
    let fee = amount * this.feeConfig.commissionRate;
    fee = Math.max(fee, this.feeConfig.minCommission);

    if (side === "sell") {
      fee += amount * this.feeConfig.stampTaxRate;
    }

    fee += amount * this.feeConfig.transferFeeRate;
    return Math.round(fee * 100) / 100;
  }

  /**
   * 创建交易记录
   */
  private createTradeRecord(
    symbol: string,
    side: "buy" | "sell",
    quantity: number,
    price: number,
    amount: number,
    fee: number,
    date: number,
    reason?: string,
  ): TradeRecord {
    const position = this.account.positions.get(symbol);
    return {
      tradeId: `T${++this.tradeCounter}`,
      symbol,
      side,
      quantity,
      price,
      amount,
      fee,
      netAmount: side === "buy" ? -(amount + fee) : amount - fee,
      tradeDate: new Date(date),
      reason,
      cashBalanceAfter: this.account.cash,
      positionQuantityAfter: position?.quantity || 0,
    };
  }

  /**
   * 创建成交订单
   */
  private createFilledOrder(
    symbol: string,
    side: "buy" | "sell",
    quantity: number,
    price: number,
    amount: number,
    fee: number,
    date: number,
    reason?: string,
  ): FilledOrder {
    return {
      orderId: `ORD_${Date.now()}_${this.tradeCounter}`,
      side,
      symbol,
      quantity,
      price,
      totalAmount: amount,
      tradeDate: new Date(date),
      fee,
      status: "filled",
      reason,
    };
  }
}
```

## 5. 组合执行器

```typescript
// packages/execution/src/portfolio-executor.ts

import {
  Account,
  Position,
  TradeRecord,
  DailySnapshot,
  ExecutionResult,
  ExecutionResultData,
  FeeConfig,
  Signal,
  DailyCandle,
  FilledOrder,
} from "@quant-backtest/core";
import { BacktestExecutor } from "./backtest-executor";

/**
 * 组合执行器
 * 支持多只股票同时执行交易
 */
export class PortfolioExecutor {
  private accounts: Map<string, Account> = new Map();
  private trades: TradeRecord[] = [];
  private snapshots: DailySnapshot[] = [];
  private feeConfig: FeeConfig;
  private tradeCounter = 0;
  private mainAccountKey = "main";

  constructor(initialCash: number, feeConfig: FeeConfig) {
    this.feeConfig = feeConfig;
    // 所有股票共用一个账户
    this.accounts.set(this.mainAccountKey, {
      initialCash,
      cash: initialCash,
      positions: new Map(),
      totalFees: 0,
      totalRealizedProfit: 0,
    });
  }

  /**
   * 执行组合信号
   */
  executePortfolioSignal(
    signals: { [symbol: string]: Signal | null },
    candles: Map<string, DailyCandle>,
  ): Map<string, ExecutionResult> {
    const results = new Map<string, ExecutionResult>();
    const account = this.accounts.get(this.mainAccountKey)!;

    for (const [symbol, signal] of Object.entries(signals)) {
      if (signal) {
        const candle = candles.get(symbol);
        if (candle) {
          const result = this.executeSingleSignal(signal, candle, account);
          results.set(symbol, result);
        }
      }
    }

    return results;
  }

  /**
   * 执行单个信号
   */
  private executeSingleSignal(
    signal: Signal,
    candle: DailyCandle,
    account: Account,
  ): ExecutionResult {
    const { action, quantity, price, reason } = signal;

    // 基本参数校验
    if (quantity <= 0) {
      return {
        success: false,
        reason: "INVALID_QUANTITY",
        message: `交易股数无效: ${quantity}，必须大于0`,
      };
    }

    if (price <= 0) {
      return {
        success: false,
        reason: "INVALID_PRICE",
        message: `交易价格无效: ${price}，必须大于0`,
      };
    }

    if (action === "buy") {
      return this.executeBuy(
        account,
        candle.symbol,
        quantity,
        price,
        candle.date,
        reason,
      );
    } else if (action === "sell") {
      return this.executeSell(
        account,
        candle.symbol,
        quantity,
        price,
        candle.date,
        reason,
      );
    } else {
      return {
        success: false,
        reason: "UNKNOWN_ERROR",
        message: `未知的交易动作: ${action}`,
      };
    }
  }

  /**
   * 执行买入（组合版本）
   */
  private executeBuy(
    account: Account,
    symbol: string,
    quantity: number,
    price: number,
    date: number,
    reason?: string,
  ): ExecutionResult {
    const amount = quantity * price;
    const fee = this.calculateFee(amount, "buy");
    const totalCost = amount + fee;

    if (account.cash < totalCost) {
      return {
        success: false,
        reason: "INSUFFICIENT_CASH",
        message: `资金不足: 需要 ${totalCost.toFixed(2)}，可用 ${account.cash.toFixed(2)}`,
      };
    }

    account.cash -= totalCost;
    account.totalFees += fee;

    const position = account.positions.get(symbol);
    let positionQuantityAfter: number;

    if (position) {
      const oldTotalCost = position.quantity * position.costPrice;
      const newTotalCost = oldTotalCost + amount;
      const newQuantity = position.quantity + quantity;
      position.costPrice = newTotalCost / newQuantity;
      position.quantity = newQuantity;
      position.totalCost = newTotalCost;
      positionQuantityAfter = newQuantity;
    } else {
      account.positions.set(symbol, {
        symbol,
        quantity,
        costPrice: price,
        totalCost: amount,
      });
      positionQuantityAfter = quantity;
    }

    const trade = this.createTradeRecord(
      symbol,
      "buy",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );
    this.trades.push(trade);

    const order = this.createFilledOrder(
      symbol,
      "buy",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );

    return { success: true, order, trade };
  }

  /**
   * 执行卖出（组合版本）
   */
  private executeSell(
    account: Account,
    symbol: string,
    quantity: number,
    price: number,
    date: number,
    reason?: string,
  ): ExecutionResult {
    const position = account.positions.get(symbol);

    if (!position) {
      return {
        success: false,
        reason: "POSITION_NOT_FOUND",
        message: `未找到股票 ${symbol} 的持仓`,
      };
    }

    if (position.quantity < quantity) {
      return {
        success: false,
        reason: "INSUFFICIENT_POSITION",
        message: `持仓不足: 需要 ${quantity}股，持有 ${position.quantity}股`,
      };
    }

    const amount = quantity * price;
    const fee = this.calculateFee(amount, "sell");
    const netRevenue = amount - fee;

    account.cash += netRevenue;
    account.totalFees += fee;

    const costOfSold = quantity * position.costPrice;
    const realizedProfit = netRevenue - costOfSold;
    account.totalRealizedProfit += realizedProfit;

    position.quantity -= quantity;
    let positionQuantityAfter: number;
    if (position.quantity === 0) {
      account.positions.delete(symbol);
      positionQuantityAfter = 0;
    } else {
      position.totalCost = position.quantity * position.costPrice;
      positionQuantityAfter = position.quantity;
    }

    const trade = this.createTradeRecord(
      symbol,
      "sell",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );
    this.trades.push(trade);

    const order = this.createFilledOrder(
      symbol,
      "sell",
      quantity,
      price,
      amount,
      fee,
      date,
      reason,
    );

    return { success: true, order, trade };
  }

  /**
   * 计算手续费
   */
  private calculateFee(amount: number, side: "buy" | "sell"): number {
    let fee = amount * this.feeConfig.commissionRate;
    fee = Math.max(fee, this.feeConfig.minCommission);

    if (side === "sell") {
      fee += amount * this.feeConfig.stampTaxRate;
    }

    fee += amount * this.feeConfig.transferFeeRate;
    return Math.round(fee * 100) / 100;
  }

  /**
   * 创建交易记录
   */
  private createTradeRecord(
    symbol: string,
    side: "buy" | "sell",
    quantity: number,
    price: number,
    amount: number,
    fee: number,
    date: number,
    reason?: string,
  ): TradeRecord {
    const account = this.accounts.get(this.mainAccountKey)!;
    const position = account.positions.get(symbol);
    return {
      tradeId: `T${++this.tradeCounter}`,
      symbol,
      side,
      quantity,
      price,
      amount,
      fee,
      netAmount: side === "buy" ? -(amount + fee) : amount - fee,
      tradeDate: new Date(date),
      reason,
      cashBalanceAfter: account.cash,
      positionQuantityAfter: position?.quantity || 0,
    };
  }

  /**
   * 创建成交订单
   */
  private createFilledOrder(
    symbol: string,
    side: "buy" | "sell",
    quantity: number,
    price: number,
    amount: number,
    fee: number,
    date: number,
    reason?: string,
  ): FilledOrder {
    return {
      orderId: `ORD_${Date.now()}_${this.tradeCounter}`,
      side,
      symbol,
      quantity,
      price,
      totalAmount: amount,
      tradeDate: new Date(date),
      fee,
      status: "filled",
      reason,
    };
  }

  /**
   * 生成快照
   */
  snapshot(date: Date, prices: Map<string, number>): void {
    const account = this.accounts.get(this.mainAccountKey)!;
    let marketValue = 0;
    const positions: any[] = [];

    for (const [symbol, pos] of account.positions) {
      const currentPrice = prices.get(symbol) || 0;
      const mv = pos.quantity * currentPrice;
      marketValue += mv;
      positions.push({
        symbol,
        quantity: pos.quantity,
        costPrice: pos.costPrice,
        totalCost: pos.totalCost,
        currentPrice,
        marketValue: mv,
        profit: mv - pos.totalCost,
        profitRate: pos.totalCost > 0 ? mv / pos.totalCost - 1 : 0,
      });
    }

    const totalAssets = account.cash + marketValue;
    const unrealizedProfit =
      marketValue -
      Array.from(account.positions.values()).reduce(
        (sum, p) => sum + p.totalCost,
        0,
      );

    this.snapshots.push({
      date,
      cash: account.cash,
      marketValue,
      totalAssets,
      realizedProfit: account.totalRealizedProfit,
      unrealizedProfit,
      totalProfit: account.totalRealizedProfit + unrealizedProfit,
      returnRate: (totalAssets - account.initialCash) / account.initialCash,
      positions,
    });
  }

  /**
   * 获取执行结果
   */
  getResult(): ExecutionResultData {
    const account = this.accounts.get(this.mainAccountKey)!;
    return {
      initialCash: account.initialCash,
      finalCash: account.cash,
      finalTotalAssets:
        this.snapshots[this.snapshots.length - 1]?.totalAssets || account.cash,
      trades: this.trades,
      snapshots: this.snapshots,
      totalFees: account.totalFees,
      totalRealizedProfit: account.totalRealizedProfit,
      finalPositions: new Map(account.positions),
    };
  }

  /**
   * 获取账户
   */
  getAccount(): Account {
    const account = this.accounts.get(this.mainAccountKey)!;
    return { ...account, positions: new Map(account.positions) };
  }
}
```

## 7. 使用示例

### 7.1 基础执行

```typescript
import { BacktestExecutor } from "@quant-backtest/execution";
import { FeeConfig } from "@quant-backtest/core";

const feeConfig: FeeConfig = {
  commissionRate: 0.00025,
  minCommission: 5,
  stampTaxRate: 0.001,
  transferFeeRate: 0.00001,
};

const executor = new BacktestExecutor(100000, feeConfig);

// 执行买入
const buyResult = executor.executeSignal(
  {
    action: "buy",
    quantity: 1000,
    price: 10.5,
    reason: "金叉买入",
  },
  {
    symbol: "510300.SH",
    date: 20241201,
    close: 10.5,
    // ... 其他字段
  },
);

if (buyResult.success) {
  console.log(
    `买入成功: ${buyResult.trade.quantity}股 @ ${buyResult.trade.price}`,
  );
} else {
  console.log(`买入失败: ${buyResult.reason} - ${buyResult.message}`);
}

// 生成快照
const prices = new Map<string, number>();
prices.set("510300.SH", 10.5);
executor.snapshot(new Date(), prices);

// 获取结果
const result = executor.getResult();
console.log(`最终资产: ${result.finalTotalAssets}`);
console.log(`总交易次数: ${result.trades.length}`);
```

import { PortfolioExecutor } from "@quant-backtest/execution";

const portfolioExecutor = new PortfolioExecutor(100000, feeConfig);
