# 06-结果与展示层

## 📖 文档概述

本文档详细描述量化回测系统的**结果层**和**展示层**，涵盖收益指标计算、风险指标计算、交易统计分析、基准对比、控制台报告输出和数据导出等核心功能。

**目标读者：** 量化分析师、前端开发工程师

## 1. 概述

### 1.1 结果层职责

| 职责             | 说明                                     |
| :--------------- | :--------------------------------------- |
| **收益指标计算** | 总收益率、年化收益率、累计盈亏           |
| **风险指标计算** | 最大回撤、夏普比率、索提诺比率、卡玛比率 |
| **交易统计分析** | 胜率、盈亏比、平均持仓天数、最大连续亏损 |
| **基准对比**     | 沪深300对比、自定义基准对比              |
| **月度收益分析** | 月度收益率分布                           |

### 1.2 展示层职责

| 职责             | 说明                       |
| :--------------- | :------------------------- |
| **控制台报告**   | 格式化输出回测报告         |
| **数据导出**     | JSON/CSV格式导出           |
| **图表数据生成** | 生成用于前端展示的图表数据 |

## 3. 结果分析器实现

```typescript
// packages/result/src/result-analyzer.ts

import {
  ExecutionResultData,
  TradeRecord,
  DailySnapshot,
  QuantResult,
  MonthlyReturn,
  BenchmarkConfig,
  BenchmarkComparison,
} from "@quant-backtest/core";

/**
 * 结果分析器
 * 计算量化指标
 */
export class ResultAnalyzer {
  protected riskFreeRate = 0.03; // 无风险利率 3%

  /**
   * 分析执行结果
   * @param result - 执行结果
   * @param initialCash - 初始本金
   * @returns 量化分析结果
   */
  analyze(result: ExecutionResultData, initialCash: number): QuantResult {
    const snapshots = result.snapshots;
    const trades = result.trades;

    if (snapshots.length === 0) {
      return this.createEmptyResult(initialCash);
    }

    const finalAssets = snapshots[snapshots.length - 1].totalAssets;
    const totalReturn = (finalAssets - initialCash) / initialCash;
    const days = snapshots.length;

    // 1. 收益指标
    const annualizedReturn = this.calculateAnnualizedReturn(totalReturn, days);

    // 2. 交易统计
    const tradeStats = this.calculateTradeStats(trades);

    // 3. 风险指标
    const riskMetrics = this.calculateRiskMetrics(snapshots, annualizedReturn);

    // 4. 持仓分析
    const holdingStats = this.calculateHoldingStats(trades);

    // 5. 月度收益
    const monthlyReturns = this.calculateMonthlyReturns(snapshots);

    return {
      // 收益指标
      totalReturn,
      totalProfit: finalAssets - initialCash,
      finalAssets,
      annualizedReturn,

      // 交易统计
      ...tradeStats,

      // 风险指标
      ...riskMetrics,

      // 持仓分析
      ...holdingStats,

      // 原始数据
      trades,
      dailySnapshots: snapshots,
      monthlyReturns,
      initialCash,
    };
  }

  /**
   * 计算年化收益率
   */
  protected calculateAnnualizedReturn(
    totalReturn: number,
    days: number,
  ): number {
    if (days === 0) return 0;
    return Math.pow(1 + totalReturn, 365 / days) - 1;
  }

  /**
   * 计算交易统计
   */
  protected calculateTradeStats(trades: TradeRecord[]): Partial<QuantResult> {
    const totalTrades = trades.length;
    if (totalTrades === 0) {
      return {
        totalTrades: 0,
        winningTrades: 0,
        losingTrades: 0,
        winRate: 0,
        avgProfit: 0,
        avgLoss: 0,
        profitLossRatio: 0,
      };
    }

    // 配对买卖单计算盈亏
    const profits = this.calculatePairProfits(trades);
    const winning = profits.filter((p) => p > 0);
    const losing = profits.filter((p) => p < 0);

    const avgProfit =
      winning.length > 0
        ? winning.reduce((a, b) => a + b, 0) / winning.length
        : 0;
    const avgLoss =
      losing.length > 0
        ? Math.abs(losing.reduce((a, b) => a + b, 0) / losing.length)
        : 0;

    return {
      totalTrades,
      winningTrades: winning.length,
      losingTrades: losing.length,
      winRate: profits.length > 0 ? winning.length / profits.length : 0,
      avgProfit,
      avgLoss,
      profitLossRatio: avgLoss > 0 ? avgProfit / avgLoss : 0,
    };
  }

  /**
   * 配对买卖单计算每笔盈亏
   */
  protected calculatePairProfits(trades: TradeRecord[]): number[] {
    const profits: number[] = [];
    let buyIndex = 0;

    while (buyIndex < trades.length) {
      // 找下一个买入
      while (buyIndex < trades.length && trades[buyIndex].side !== "buy") {
        buyIndex++;
      }
      if (buyIndex >= trades.length) break;

      // 找对应的卖出
      let sellIndex = buyIndex + 1;
      while (sellIndex < trades.length && trades[sellIndex].side !== "sell") {
        sellIndex++;
      }
      if (sellIndex >= trades.length) break;

      const buy = trades[buyIndex];
      const sell = trades[sellIndex];
      profits.push((sell.price - buy.price) * buy.quantity);
      buyIndex = sellIndex + 1;
    }

    return profits;
  }

  /**
   * 计算风险指标
   */
  protected calculateRiskMetrics(
    snapshots: DailySnapshot[],
    annualizedReturn: number,
  ): Partial<QuantResult> {
    // 计算每日收益率
    const returns: number[] = [];
    for (let i = 1; i < snapshots.length; i++) {
      const prev = snapshots[i - 1].totalAssets;
      const curr = snapshots[i].totalAssets;
      if (prev > 0) {
        returns.push((curr - prev) / prev);
      }
    }

    // 最大回撤
    let maxDrawdown = 0;
    let maxDrawdownDuration = 0;
    let peak = snapshots[0].totalAssets;
    let drawdownStart = 0;
    let currentDrawdown = 0;

    for (let i = 0; i < snapshots.length; i++) {
      if (snapshots[i].totalAssets > peak) {
        peak = snapshots[i].totalAssets;
        drawdownStart = i;
        currentDrawdown = 0;
      } else {
        currentDrawdown = (peak - snapshots[i].totalAssets) / peak;
        maxDrawdown = Math.max(maxDrawdown, currentDrawdown);
        maxDrawdownDuration = Math.max(maxDrawdownDuration, i - drawdownStart);
      }
    }

    // 夏普比率
    const sharpeRatio = this.calculateSharpeRatio(returns, annualizedReturn);

    // 索提诺比率
    const sortinoRatio = this.calculateSortinoRatio(returns, annualizedReturn);

    // 卡玛比率
    const calmarRatio = maxDrawdown > 0 ? annualizedReturn / maxDrawdown : 0;

    return {
      maxDrawdown,
      maxDrawdownDuration,
      sharpeRatio,
      sortinoRatio,
      calmarRatio,
    };
  }

  /**
   * 计算夏普比率
   */
  protected calculateSharpeRatio(
    returns: number[],
    annualizedReturn: number,
  ): number {
    if (returns.length === 0) return 0;
    const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
    const stdDev = this.calculateStdDev(returns);
    if (stdDev === 0) return 0;
    return (annualizedReturn - this.riskFreeRate) / (stdDev * Math.sqrt(365));
  }

  /**
   * 计算索提诺比率
   */
  protected calculateSortinoRatio(
    returns: number[],
    annualizedReturn: number,
  ): number {
    if (returns.length === 0) return 0;
    const negativeReturns = returns.filter((r) => r < 0);
    if (negativeReturns.length === 0) return 0;
    const downsideDeviation = this.calculateStdDev(negativeReturns);
    if (downsideDeviation === 0) return 0;
    return (
      (annualizedReturn - this.riskFreeRate) /
      (downsideDeviation * Math.sqrt(365))
    );
  }

  /**
   * 计算标准差
   */
  protected calculateStdDev(values: number[]): number {
    if (values.length === 0) return 0;
    const mean = values.reduce((a, b) => a + b, 0) / values.length;
    const squaredDiffs = values.map((v) => Math.pow(v - mean, 2));
    return Math.sqrt(squaredDiffs.reduce((a, b) => a + b, 0) / values.length);
  }

  /**
   * 计算持仓统计
   */
  protected calculateHoldingStats(trades: TradeRecord[]): Partial<QuantResult> {
    // 配对买卖计算持仓天数
    const holdingDays: number[] = [];
    let buyIndex = 0;

    while (buyIndex < trades.length) {
      while (buyIndex < trades.length && trades[buyIndex].side !== "buy") {
        buyIndex++;
      }
      if (buyIndex >= trades.length) break;

      let sellIndex = buyIndex + 1;
      while (sellIndex < trades.length && trades[sellIndex].side !== "sell") {
        sellIndex++;
      }
      if (sellIndex >= trades.length) break;

      const buy = trades[buyIndex];
      const sell = trades[sellIndex];
      const days =
        (sell.tradeDate.getTime() - buy.tradeDate.getTime()) /
        (1000 * 60 * 60 * 24);
      holdingDays.push(days);
      buyIndex = sellIndex + 1;
    }

    const avgHoldingDays =
      holdingDays.length > 0
        ? holdingDays.reduce((a, b) => a + b, 0) / holdingDays.length
        : 0;

    // 计算最大连续亏损
    const profits = this.calculatePairProfits(trades);
    let maxConsecutive = 0;
    let current = 0;
    for (const p of profits) {
      if (p < 0) {
        current++;
        maxConsecutive = Math.max(maxConsecutive, current);
      } else {
        current = 0;
      }
    }

    return {
      avgHoldingDays,
      maxConsecutiveLosses: maxConsecutive,
    };
  }

  /**
   * 计算月度收益
   */
  protected calculateMonthlyReturns(
    snapshots: DailySnapshot[],
  ): MonthlyReturn[] {
    const monthlyMap = new Map<string, { start: number; end: number }>();

    for (const s of snapshots) {
      const key = `${s.date.getFullYear()}-${String(s.date.getMonth() + 1).padStart(2, "0")}`;
      if (!monthlyMap.has(key)) {
        monthlyMap.set(key, { start: s.totalAssets, end: s.totalAssets });
      } else {
        monthlyMap.get(key)!.end = s.totalAssets;
      }
    }

    const result: MonthlyReturn[] = [];
    for (const [key, { start, end }] of monthlyMap) {
      const [year, month] = key.split("-").map(Number);
      result.push({
        year,
        month,
        returnRate: start > 0 ? (end - start) / start : 0,
        totalAssets: end,
      });
    }

    return result.sort((a, b) => a.year - b.year || a.month - b.month);
  }

  /**
   * 创建空结果
   */
  protected createEmptyResult(initialCash: number): QuantResult {
    return {
      totalReturn: 0,
      totalProfit: 0,
      finalAssets: initialCash,
      annualizedReturn: 0,
      totalTrades: 0,
      winningTrades: 0,
      losingTrades: 0,
      winRate: 0,
      avgProfit: 0,
      avgLoss: 0,
      profitLossRatio: 0,
      maxDrawdown: 0,
      maxDrawdownDuration: 0,
      sharpeRatio: 0,
      sortinoRatio: 0,
      calmarRatio: 0,
      avgHoldingDays: 0,
      maxConsecutiveLosses: 0,
      trades: [],
      dailySnapshots: [],
      monthlyReturns: [],
      initialCash,
    };
  }
}
```

## 4. 基准对比分析器

```typescript
// packages/result/src/enhanced-result-analyzer.ts

import {
  ExecutionResultData,
  DailySnapshot,
  QuantResult,
  BenchmarkConfig,
  BenchmarkComparison,
  MonthlyReturn,
} from "@quant-backtest/core";
import { ResultAnalyzer } from "./result-analyzer";

/**
 * 增强版结果分析器
 * 包含基准对比能力
 */
export class EnhancedResultAnalyzer extends ResultAnalyzer {
  /**
   * 分析执行结果（含基准对比）
   */
  analyzeWithBenchmark(
    result: ExecutionResultData,
    initialCash: number,
    benchmarkConfig?: BenchmarkConfig,
  ): QuantResult {
    // 1. 基础分析
    const baseResult = this.analyze(result, initialCash);

    // 2. 如果没有基准配置，返回基础结果
    if (!benchmarkConfig || benchmarkConfig.data.length === 0) {
      return baseResult;
    }

    // 3. 对齐日期
    const alignedData = this.alignDates(result.snapshots, benchmarkConfig.data);

    // 4. 计算基准对比指标
    const comparison = this.calculateBenchmarkComparison(
      alignedData.strategyValues,
      alignedData.benchmarkValues,
      benchmarkConfig.name,
      result.snapshots.length,
    );

    baseResult.benchmarkComparison = comparison;
    return baseResult;
  }

  /**
   * 对齐策略和基准的日期
   */
  private alignDates(
    snapshots: DailySnapshot[],
    benchmarkData: Array<{ date: Date; value: number }>,
  ): {
    strategyValues: number[];
    benchmarkValues: number[];
    dates: Date[];
  } {
    const strategyMap = new Map<string, number>();
    for (const s of snapshots) {
      const key = s.date.toISOString().split("T")[0];
      strategyMap.set(key, s.totalAssets);
    }

    const benchmarkMap = new Map<string, number>();
    for (const b of benchmarkData) {
      const key = b.date.toISOString().split("T")[0];
      benchmarkMap.set(key, b.value);
    }

    const dates: Date[] = [];
    const strategyValues: number[] = [];
    const benchmarkValues: number[] = [];

    for (const s of snapshots) {
      const key = s.date.toISOString().split("T")[0];
      const benchmarkValue = benchmarkMap.get(key);
      if (benchmarkValue !== undefined) {
        dates.push(s.date);
        strategyValues.push(s.totalAssets);
        benchmarkValues.push(benchmarkValue);
      }
    }

    return { strategyValues, benchmarkValues, dates };
  }

  /**
   * 计算基准对比指标
   */
  private calculateBenchmarkComparison(
    strategyValues: number[],
    benchmarkValues: number[],
    benchmarkName: string,
    totalDays: number,
  ): BenchmarkComparison {
    const initialStrategy = strategyValues[0] || 1;
    const initialBenchmark = benchmarkValues[0] || 1;

    // 总收益率
    const strategyReturn =
      strategyValues[strategyValues.length - 1] / initialStrategy - 1;
    const benchmarkReturn =
      benchmarkValues[benchmarkValues.length - 1] / initialBenchmark - 1;

    // 超额收益（Alpha）
    const alpha = strategyReturn - benchmarkReturn;
    const annualizedAlpha = Math.pow(1 + alpha, 365 / totalDays) - 1;

    // 计算Beta
    const { beta } = this.calculateBeta(strategyValues, benchmarkValues);

    // 计算月度超额收益
    const monthlyAlpha = this.calculateMonthlyAlpha(
      strategyValues,
      benchmarkValues,
    );

    // 计算跟踪误差和信息比率
    const dailyReturns = this.calculateDailyReturns(strategyValues);
    const benchmarkDailyReturns = this.calculateDailyReturns(benchmarkValues);
    const alphaDaily = dailyReturns.map(
      (r, i) => r - (benchmarkDailyReturns[i] || 0),
    );

    const trackingError = this.calculateStdDev(alphaDaily) * Math.sqrt(252);
    const avgAlpha = alphaDaily.reduce((a, b) => a + b, 0) / alphaDaily.length;
    const informationRatio =
      trackingError > 0 ? (avgAlpha * Math.sqrt(252)) / trackingError : 0;

    return {
      benchmarkName,
      strategyReturn,
      benchmarkReturn,
      alpha,
      annualizedAlpha,
      beta,
      informationRatio,
      trackingError,
      monthlyAlpha,
    };
  }

  /**
   * 计算Beta系数
   */
  private calculateBeta(
    strategyValues: number[],
    benchmarkValues: number[],
  ): { beta: number; alpha: number } {
    const strategyReturns: number[] = [];
    const benchmarkReturns: number[] = [];

    for (let i = 1; i < strategyValues.length; i++) {
      const sReturn =
        (strategyValues[i] - strategyValues[i - 1]) / strategyValues[i - 1];
      const bReturn =
        (benchmarkValues[i] - benchmarkValues[i - 1]) / benchmarkValues[i - 1];
      strategyReturns.push(sReturn);
      benchmarkReturns.push(bReturn);
    }

    const n = benchmarkReturns.length;
    if (n === 0) return { beta: 1, alpha: 0 };

    const sumX = benchmarkReturns.reduce((a, b) => a + b, 0);
    const sumY = strategyReturns.reduce((a, b) => a + b, 0);
    const sumXY = benchmarkReturns.reduce(
      (a, b, i) => a + b * strategyReturns[i],
      0,
    );
    const sumX2 = benchmarkReturns.reduce((a, b) => a + b * b, 0);

    const beta = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
    const alpha = (sumY - beta * sumX) / n;

    return { beta, alpha };
  }

  /**
   * 计算月度超额收益
   */
  private calculateMonthlyAlpha(
    strategyValues: number[],
    benchmarkValues: number[],
  ): Array<{ year: number; month: number; alpha: number }> {
    const monthlyAlpha: Array<{ year: number; month: number; alpha: number }> =
      [];
    const months = Math.floor(strategyValues.length / 21);

    for (let m = 0; m < months; m++) {
      const startIdx = m * 21;
      const endIdx = Math.min((m + 1) * 21 - 1, strategyValues.length - 1);

      const sStart = strategyValues[startIdx];
      const sEnd = strategyValues[endIdx];
      const bStart = benchmarkValues[startIdx];
      const bEnd = benchmarkValues[endIdx];

      const sReturn = (sEnd - sStart) / sStart;
      const bReturn = (bEnd - bStart) / bStart;

      const date = new Date();
      monthlyAlpha.push({
        year: date.getFullYear(),
        month: m + 1,
        alpha: sReturn - bReturn,
      });
    }

    return monthlyAlpha;
  }

  private calculateDailyReturns(values: number[]): number[] {
    const returns: number[] = [];
    for (let i = 1; i < values.length; i++) {
      returns.push((values[i] - values[i - 1]) / values[i - 1]);
    }
    return returns;
  }
}
```

```typescript
// packages/presentation/src/console-presenter.ts

import {
  QuantResult,
  BenchmarkComparison,
  TradeRecord,
  DailySnapshot,
  MonthlyReturn,
} from "@quant-backtest/core";

/**
 * 报告配置
 */
export interface ReportConfig {
  /** 显示详细交易列表 */
  showTrades?: boolean;
  /** 显示每日快照 */
  showSnapshots?: boolean;
  /** 数字精度（小数位数） */
  precision?: number;
  /** 日期格式 */
  dateFormat?: string;
}

/**
 * 控制台展示器
 * 格式化输出回测报告
 */
export class ConsolePresenter {
  /**
   * 打印回测报告
   */
  printReport(result: QuantResult, config: ReportConfig = {}): void {
    const { precision = 2, showTrades = false, showSnapshots = false } = config;

    console.log("\n" + "=".repeat(80));
    console.log("                    回测报告");
    console.log("=".repeat(80));

    // 1. 基本信息
    this.printBasicInfo(result);

    // 2. 收益指标
    this.printReturnMetrics(result, precision);

    // 3. 交易统计
    this.printTradeStats(result, precision);

    // 4. 风险指标
    this.printRiskMetrics(result, precision);

    // 5. 基准对比（如果有）
    if (result.benchmarkComparison) {
      this.printBenchmarkComparison(result.benchmarkComparison, precision);
    }

    // 6. 月度收益
    this.printMonthlyReturns(result.monthlyReturns, precision);

    // 7. 交易明细（可选）
    if (showTrades) {
      this.printTrades(result.trades);
    }

    console.log("=".repeat(80) + "\n");
  }

  /**
   * 打印基本信息
   */
  private printBasicInfo(result: QuantResult): void {
    const startDate = result.dailySnapshots[0]?.date;
    const endDate =
      result.dailySnapshots[result.dailySnapshots.length - 1]?.date;
    const days = result.dailySnapshots.length;

    console.log(
      "\n----------------------------------------------------------------",
    );
    console.log("【基本信息】");
    console.log(`  初始资金: ${this.formatCurrency(result.initialCash)}`);
    console.log(`  最终资产: ${this.formatCurrency(result.finalAssets)}`);
    if (startDate && endDate) {
      console.log(
        `  回测周期: ${startDate.toISOString().split("T")[0]} ~ ${endDate.toISOString().split("T")[0]} (${days}个交易日)`,
      );
    }
  }

  /**
   * 打印收益指标
   */
  private printReturnMetrics(result: QuantResult, precision: number): void {
    console.log(
      "\n----------------------------------------------------------------",
    );
    console.log("【收益指标】");
    console.log(
      `  总收益率: ${this.formatPercent(result.totalReturn, precision)}`,
    );
    console.log(
      `  年化收益率: ${this.formatPercent(result.annualizedReturn, precision)}`,
    );
    console.log(`  总盈亏: ${this.formatCurrency(result.totalProfit)}`);
  }

  /**
   * 打印交易统计
   */
  private printTradeStats(result: QuantResult, precision: number): void {
    console.log(
      "\n----------------------------------------------------------------",
    );
    console.log("【交易统计】");
    console.log(`  总交易次数: ${result.totalTrades}`);
    if (result.totalTrades > 0) {
      console.log(
        `  盈利交易: ${result.winningTrades} (${this.formatPercent(result.winRate, precision)})`,
      );
      console.log(
        `  亏损交易: ${result.losingTrades} (${this.formatPercent(1 - result.winRate, precision)})`,
      );
      console.log(`  平均盈利: ${this.formatCurrency(result.avgProfit)}`);
      console.log(`  平均亏损: ${this.formatCurrency(result.avgLoss)}`);
      console.log(`  盈亏比: ${result.profitLossRatio.toFixed(2)}`);
    }
  }

  /**
   * 打印风险指标
   */
  private printRiskMetrics(result: QuantResult, precision: number): void {
    console.log(
      "\n----------------------------------------------------------------",
    );
    console.log("【风险指标】");
    console.log(
      `  最大回撤: ${this.formatPercent(result.maxDrawdown, precision)}`,
    );
    console.log(`  最大回撤持续: ${result.maxDrawdownDuration}天`);
    console.log(`  夏普比率: ${result.sharpeRatio.toFixed(2)}`);
    console.log(`  索提诺比率: ${result.sortinoRatio.toFixed(2)}`);
    console.log(`  卡玛比率: ${result.calmarRatio.toFixed(2)}`);
    console.log(`  平均持仓天数: ${result.avgHoldingDays.toFixed(1)}天`);
    console.log(`  最大连续亏损: ${result.maxConsecutiveLosses}次`);
  }

  /**
   * 打印基准对比
   */
  private printBenchmarkComparison(
    comparison: BenchmarkComparison,
    precision: number,
  ): void {
    console.log(
      "\n----------------------------------------------------------------",
    );
    console.log("【基准对比】");
    console.log(`  基准: ${comparison.benchmarkName}`);
    console.log(
      `  策略收益: ${this.formatPercent(comparison.strategyReturn, precision)}`,
    );
    console.log(
      `  基准收益: ${this.formatPercent(comparison.benchmarkReturn, precision)}`,
    );
    console.log(
      `  超额收益: ${this.formatPercent(comparison.alpha, precision)}`,
    );
    console.log(
      `  年化超额: ${this.formatPercent(comparison.annualizedAlpha, precision)}`,
    );
    console.log(`  Beta系数: ${comparison.beta.toFixed(3)}`);
    console.log(`  信息比率: ${comparison.informationRatio.toFixed(2)}`);
    console.log(
      `  跟踪误差: ${this.formatPercent(comparison.trackingError, precision)}`,
    );
  }

  /**
   * 打印月度收益
   */
  private printMonthlyReturns(
    monthlyReturns: MonthlyReturn[],
    precision: number,
  ): void {
    console.log(
      "\n----------------------------------------------------------------",
    );
    console.log("【月度收益】");
    const recent = monthlyReturns.slice(-12);
    for (const m of recent) {
      const sign = m.returnRate >= 0 ? "+" : "";
      console.log(
        `  ${m.year}-${String(m.month).padStart(2, "0")}: ${sign}${this.formatPercent(m.returnRate, precision)}`,
      );
    }
  }

  /**
   * 打印交易明细
   */
  private printTrades(trades: TradeRecord[]): void {
    console.log(
      "\n----------------------------------------------------------------",
    );
    console.log("【交易明细】");
    if (trades.length === 0) {
      console.log("  无交易记录");
      return;
    }

    const recent = trades.slice(-20);
    console.log(`  最近${recent.length}笔交易:`);
    console.log("  日期\t\t方向\t股数\t价格\t金额\t理由");
    for (const t of recent) {
      const date = t.tradeDate.toISOString().split("T")[0];
      const side = t.side === "buy" ? "买入" : "卖出";
      console.log(
        `  ${date}\t${side}\t${t.quantity}\t${t.price.toFixed(2)}\t${this.formatCurrency(t.amount)}\t${t.reason || "-"}`,
      );
    }
  }

  /**
   * 格式化百分比
   */
  private formatPercent(value: number, precision: number): string {
    return `${(value * 100).toFixed(precision)}%`;
  }

  /**
   * 格式化货币
   */
  private formatCurrency(value: number): string {
    return value.toLocaleString("zh-CN", {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    });
  }

  /**
   * 导出数据
   */
  exportData(result: QuantResult, format: "json" | "csv"): void {
    if (format === "json") {
      console.log(JSON.stringify(result, null, 2));
    } else if (format === "csv") {
      // CSV导出 - 交易明细
      console.log(
        "tradeId,symbol,side,quantity,price,amount,fee,netAmount,tradeDate,reason",
      );
      for (const t of result.trades) {
        console.log(
          `${t.tradeId},${t.symbol},${t.side},${t.quantity},${t.price.toFixed(4)},${t.amount.toFixed(2)},${t.fee.toFixed(2)},${t.netAmount.toFixed(2)},${t.tradeDate.toISOString()},${t.reason || ""}`,
        );
      }
    }
  }

  /**
   * 生成图表数据（用于前端展示）
   */
  generateChartData(result: QuantResult): any {
    return {
      equityCurve: result.dailySnapshots.map((s) => ({
        date: s.date,
        value: s.totalAssets / result.initialCash,
      })),
      drawdown: result.dailySnapshots.map((s, i, arr) => {
        const peak = Math.max(...arr.slice(0, i + 1).map((a) => a.totalAssets));
        return {
          date: s.date,
          value: (peak - s.totalAssets) / peak,
        };
      }),
      monthlyReturns: result.monthlyReturns,
      benchmarkComparison: result.benchmarkComparison,
    };
  }
}
```

## 6. 使用示例

### 6.1 基础分析

```typescript
import { BacktestExecutor } from "@quant-backtest/execution";
import { ResultAnalyzer } from "@quant-backtest/result";
import { ConsolePresenter } from "@quant-backtest/presentation";
import { FeeConfig } from "@quant-backtest/core";

const feeConfig: FeeConfig = {
  commissionRate: 0.00025,
  minCommission: 5,
  stampTaxRate: 0.001,
  transferFeeRate: 0.00001,
};

// 1. 执行回测
const executor = new BacktestExecutor(100000, feeConfig);
// ... 执行交易 ...

// 2. 获取结果
const executionResult = executor.getResult();

// 3. 分析结果
const analyzer = new ResultAnalyzer();
const quantResult = analyzer.analyze(executionResult, 100000);

// 4. 展示结果
const presenter = new ConsolePresenter();
presenter.printReport(quantResult, { showTrades: true });
```

### 6.2 基准对比分析

```typescript
import { EnhancedResultAnalyzer } from "@quant-backtest/result";

const enhancedAnalyzer = new EnhancedResultAnalyzer();

// 加载基准数据（沪深300）
const benchmarkConfig = {
  name: "沪深300",
  data: hs300Data.map((d) => ({

  })),
};

const quantResult = enhancedAnalyzer.analyzeWithBenchmark(
  executionResult,
  100000,
  benchmarkConfig,
);

presenter.printReport(quantResult);
```

### 6.3 导出数据

```typescript
// 导出JSON
presenter.exportData(quantResult, "json");

// 导出CSV（交易明细）
presenter.exportData(quantResult, "csv");

// 生成图表数据
const chartData = presenter.generateChartData(quantResult);
console.log("图表数据:", JSON.stringify(chartData, null, 2));
```

## 7. 指标说明

| 指标           | 计算公式                                 | 说明                 |
| :------------- | :--------------------------------------- | :------------------- |
| **总收益率**   | `(最终资产 - 初始本金) / 初始本金`       | 回测期间总收益       |
| **年化收益率** | `(1 + 总收益率)^(365/天数) - 1`          | 年化后的收益率       |
| **胜率**       | `盈利交易次数 / 总交易次数`              | 赚钱交易的比例       |
| **盈亏比**     | `平均盈利 / 平均亏损`                    | 赚钱与亏钱的平均比例 |
| **最大回撤**   | `max((峰值 - 谷值) / 峰值)`              | 最大亏损幅度         |
| **夏普比率**   | `(年化收益率 - 无风险利率) / 年化波动率` | 风险调整后收益       |
| **索提诺比率** | `(年化收益率 - 无风险利率) / 下行波动率` | 只考虑下行风险       |
| **卡玛比率**   | `年化收益率 / 最大回撤`                  | 收益与最大回撤的比率 |
